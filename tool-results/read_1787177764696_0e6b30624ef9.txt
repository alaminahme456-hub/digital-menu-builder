    78→// ─── Step Indicator ────────────────────────────────────────────────
    79→function StepIndicator({ steps, current }: { steps: string[]; current: number }) {
    80→  return (
    81→    <div className="flex items-center gap-1 overflow-x-auto pb-2">
    82→      {steps.map((s, i) => (
    83→        <React.Fragment key={i}>
    84→          <div className="flex shrink-0 items-center gap-2">
    85→            <div className={`flex size-8 items-center justify-center rounded-full text-xs font-bold transition-colors ${i <= current ? 'bg-charcoal text-white' : 'bg-slate-200 text-slate-500'}`}>
    86→              {i < current ? <Check className="size-4" /> : i + 1}
    87→            </div>
    88→            <span className={`hidden text-sm sm:inline ${i <= current ? 'text-charcoal font-medium' : 'text-slate-400'}`}>{s}</span>
    89→          </div>
    90→          {i < steps.length - 1 && <div className={`mx-1 h-px w-6 shrink-0 ${i < current ? 'bg-charcoal' : 'bg-slate-200'}`} />}
    91→        </React.Fragment>
    92→      ))}
    93→    </div>
    94→  );
    95→}
