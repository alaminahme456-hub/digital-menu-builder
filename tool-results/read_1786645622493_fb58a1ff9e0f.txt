   322→export default function Templates() {
   323→  const { token } = useAuthStore();
   324→  const { currentBusiness } = useAppStore();
   325→
   326→  const [previewTemplate, setPreviewTemplate] = useState<string | null>(null);
   327→  const [applying, setApplying] = useState(false);
   328→  const [saving, setSaving] = useState(false);
   329→  const [primaryColor, setPrimaryColor] = useState('#10b981');
   330→  const [secondaryColor, setSecondaryColor] = useState('#f1f5f9');
   331→  const [selectedFont, setSelectedFont] = useState('inter');
   332→
   333→  // Sync with current business data
   334→  useEffect(() => {
   335→    if (currentBusiness) {
   336→      setPrimaryColor(currentBusiness.primaryColor || '#10b981');
   337→      setSecondaryColor(currentBusiness.secondaryColor || '#f1f5f9');
   338→      setSelectedFont(currentBusiness.fontFamily || 'inter');
   339→    }
   340→  }, [currentBusiness]);
   341→
   342→  const currentTemplate = currentBusiness?.templateName || 'modern';
   343→
   344→  const applyTemplate = async (templateName: string) => {
   345→    if (!currentBusiness?.id || !token) return;
   346→
   347→    setApplying(true);
   348→    try {
   349→      const res = await fetch(`/api/businesses/${currentBusiness.id}`, {
   350→        method: 'PUT',
   351→        headers: {
   352→          Authorization: `Bearer ${token}`,
   353→          'Content-Type': 'application/json',
   354→        },
   355→        body: JSON.stringify({ templateName }),
   356→      });
   357→
   358→      if (res.ok) {
   359→        toast.success(`"${TEMPLATES.find(t => t.name === templateName)?.label}" template applied!`);
   360→      } else {
   361→        toast.error('Failed to apply template');
   362→      }
   363→    } catch {
   364→      toast.error('Failed to apply template');

   366→      setApplying(false);
   367→      setPreviewTemplate(null);
   368→    }
   369→  };

   377→        method: 'PUT',

   379→          Authorization: `Bearer ${token}`,
   380→          'Content-Type': 'application/json',

   382→        body: JSON.stringify({
   383→          primaryColor,
   384→          secondaryColor,
   385→          fontFamily: selectedFont,
   386→        }),
   387→      });
   388→
   389→      if (res.ok) {
   390→        toast.success('Customization applied successfully!');
   391→      } else {
   392→        toast.error('Failed to apply changes');
   393→      }
   394→    } catch {
   395→      toast.error('Failed to apply changes');
   396→    } finally {
   397→      setSaving(false);
   398→    }
   399→  };
   400→
   401→  if (!currentBusiness) {
   402→    return (
   403→      <Card>
   404→        <CardContent className="flex flex-col items-center justify-center py-16">

   408→      </Card>
   409→    );
   410→  }
   411→

   413→    <div className="space-y-6">

   416→        <CardHeader>
   417→          <CardTitle className="flex items-center gap-2">
