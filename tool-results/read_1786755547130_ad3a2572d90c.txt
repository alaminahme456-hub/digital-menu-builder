   359→// ═══════════════════════════════════════════
   360→// MAIN MARKETPLACE COMPONENT
   361→// ═══════════════════════════════════════════
   362→export default function Marketplace() {
   363→  const { token, user } = useAuthStore();
   364→  const {
   365→    currentBusiness, marketplaceFavorites, toggleMarketplaceFavorite,
   366→    appliedMarketplaceTemplate, appliedMarketplaceCoverTemplate,
   367→    setAppliedMarketplaceTemplate, setAppliedMarketplaceCoverTemplate,
   368→    navigate,
   369→  } = useAppStore();
   370→
   371→  // Tab & UI state
   372→  const [activeTab, setActiveTab] = useState('discover');
   373→  const [search, setSearch] = useState('');
   374→  const [filterType, setFilterType] = useState('all');
   375→  const [filterCategory, setFilterCategory] = useState('All');
   376→  const [sortBy, setSortBy] = useState('popular');
   377→  const [page, setPage] = useState(1);
   378→  const [hasMore, setHasMore] = useState(true);
   379→
   380→  // Data state
   381→  const [templates, setTemplates] = useState<MarketplaceTemplate[]>([]);
   382→  const [featured, setFeatured] = useState<MarketplaceTemplate[]>([]);
   383→  const [designers, setDesigners] = useState<MarketplaceDesigner[]>([]);
   384→  const [favorites, setFavorites] = useState<MarketplaceTemplate[]>([]);
   385→  const [loading, setLoading] = useState(true);
   386→  const [loadingMore, setLoadingMore] = useState(false);
   387→  const [designersLoading, setDesignersLoading] = useState(false);
   388→  const [favoritesLoading, setFavoritesLoading] = useState(false);
   389→  const [applyingId, setApplyingId] = useState<string | null>(null);
   390→
   391→  // ── Fetch Templates ──
   392→  const fetchTemplates = useCallback(async (append = false) => {
   393→    if (!append) setLoading(true);
   394→    else setLoadingMore(true);
   395→
   396→    try {
   397→      const params = new URLSearchParams();
   398→      if (activeTab === 'book_covers') params.set('type', 'book_cover');
   399→      else if (activeTab === 'menu') params.set('type', 'menu');
   400→      else if (filterType !== 'all') params.set('type', filterType);
   401→      if (filterCategory !== 'All') params.set('category', filterCategory);
   402→      if (search.trim()) params.set('search', search.trim());
   403→      params.set('sort', sortBy);
   404→      params.set('page', String(append ? page : 1));
   405→      params.set('limit', '20');
   406→
   407→      const res = await fetch(`/api/marketplace/templates?${params}`, {
   408→        headers: token ? { Authorization: `Bearer ${token}` } : {},
   409→      });
   410→      if (res.ok) {
   411→        const data = await res.json();
   412→        const list = data.templates || data || [];
   413→        setTemplates(append ? (prev) => [...prev, ...list] : list);
   414→        setHasMore(list.length >= 20);
   415→        if (!append) setPage(2);
   416→        else setPage((p) => p + 1);
   417→      }
   418→    } catch {
   419→      toast.error('Failed to load templates');
   420→    } finally {
   421→      setLoading(false);
   422→      setLoadingMore(false);
   423→    }
   424→  }, [activeTab, filterType, filterCategory, search, sortBy, page, token]);
   425→
   426→  // ── Fetch Featured ──
   427→  const fetchFeatured = useCallback(async () => {
   428→    try {
   429→      const res = await fetch('/api/marketplace/templates?featured=true&limit=10', {
   430→        headers: token ? { Authorization: `Bearer ${token}` } : {},
   431→      });
   432→      if (res.ok) {
   433→        const data = await res.json();
   434→        setFeatured(data.templates || data || []);
   435→      }
   436→    } catch { /* silent */ }
   437→  }, [token]);
   438→
   439→  // ── Fetch Designers ──
   440→  const fetchDesigners = useCallback(async (searchQuery = '') => {
   441→    setDesignersLoading(true);
   442→    try {
   443→      const params = new URLSearchParams({ limit: '50' });
   444→      if (searchQuery) params.set('search', searchQuery);
   445→      const res = await fetch(`/api/marketplace/designers?${params}`, {
   446→        headers: token ? { Authorization: `Bearer ${token}` } : {},
   447→      });
   448→      if (res.ok) {
   449→        const data = await res.json();
   450→        setDesigners(data.designers || data || []);
   451→      }
   452→    } catch {
   453→      toast.error('Failed to load designers');
   454→    } finally {
   455→      setDesignersLoading(false);
   456→    }
   457→  }, [token]);
   458→
   459→  // ── Fetch Favorites ──
   460→  const fetchFavorites = useCallback(async () => {
   461→    setFavoritesLoading(true);
   462→    try {
   463→      const res = await fetch('/api/marketplace/favorites', {
   464→        headers: token ? { Authorization: `Bearer ${token}` } : {},
   465→      });
   466→      if (res.ok) {
   467→        const data = await res.json();
   468→        setFavorites(data.templates || data || []);
   469→      }
   470→    } catch {
   471→      toast.error('Failed to load favorites');
   472→    } finally {
   473→      setFavoritesLoading(false);
   474→    }
   475→  }, [token]);
   476→
   477→  // ── Effects ──
   478→  useEffect(() => { fetchFeatured(); }, [fetchFeatured]);
   479→
   480→  useEffect(() => {
   481→    setPage(1);
   482→    fetchTemplates(false);
   483→  }, [activeTab, filterType, filterCategory, search, sortBy]);
   484→
   485→  useEffect(() => {
   486→    if (activeTab === 'designers') fetchDesigners(search);
   487→  }, [activeTab, search, fetchDesigners]);
   488→
   489→  useEffect(() => {
   490→    if (activeTab === 'favorites') fetchFavorites();
   491→  }, [activeTab, fetchFavorites]);
   492→
   493→  // ── Toggle Favorite ──
   494→  const handleToggleFavorite = useCallback(async (id: string) => {
   495→    const wasFav = marketplaceFavorites.includes(id);
   496→    toggleMarketplaceFavorite(id);
   497→    try {
   498→      await fetch('/api/marketplace/favorites', {
   499→        method: 'POST',
   500→        headers: {
   501→          'Content-Type': 'application/json',
   502→          ...(token ? { Authorization: `Bearer ${token}` } : {}),
   503→        },
   504→        body: JSON.stringify({ templateId: id, action: wasFav ? 'remove' : 'add' }),
   505→      });
   506→      if (activeTab === 'favorites') fetchFavorites();
   507→    } catch {
   508→      toggleMarketplaceFavorite(id);
   509→      toast.error('Failed to update favorites');
   510→    }
   511→  }, [marketplaceFavorites, toggleMarketplaceFavorite, token, activeTab, fetchFavorites]);
   512→
   513→  // ── Apply Template ──
   514→  const handleApply = useCallback(async (template: MarketplaceTemplate) => {
   515→    if (!currentBusiness?.id) {
   516→      toast.error('Please select a business first');
   517→      return;
   518→    }
   519→
   520→    setApplyingId(template.id);
   521→    const prevMenu = appliedMarketplaceTemplate;
   522→    const prevCover = appliedMarketplaceCoverTemplate;
   523→
   524→    // Optimistic
   525→    if (template.template_type === 'menu') setAppliedMarketplaceTemplate(template.id);
   526→    else setAppliedMarketplaceCoverTemplate(template.id);
