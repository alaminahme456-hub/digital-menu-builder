    42→/*  Template helpers                                                   */
    43→/* ------------------------------------------------------------------ */
    44→type TemplateName =
    45→  | 'modern' | 'classic' | 'luxury' | 'minimal' | 'fastfood'
    46→  | 'cafe' | 'pizza' | 'dark' | 'colorful' | 'elegant';
    47→
    48→function getTemplateClasses(template: string, primaryColor: string): string {
    49→  const t = template as TemplateName;
    50→  const base = 'min-h-screen';
    51→  const overrides: Record<TemplateName, string> = {
    52→    modern: 'bg-white text-gray-900',
    53→    classic: 'bg-amber-50/50 text-gray-900 font-serif',
    54→    luxury: 'bg-gray-950 text-gray-100',
    55→    minimal: 'bg-white text-gray-900',
    56→    fastfood: 'bg-yellow-50 text-gray-900',
    57→    cafe: 'bg-orange-50/40 text-gray-900',
    58→    pizza: 'bg-red-50/30 text-gray-900',
    59→    dark: 'bg-[#1a1a2e] text-gray-100',
    60→    colorful: 'bg-white text-gray-900',
    61→    elegant: 'bg-stone-50 text-gray-800',
    62→  };
    63→  return `${base} ${overrides[t] ?? overrides.modern}`;
    64→}
    65→
    66→function getHeaderBg(template: string, primaryColor: string): string {
    67→  const t = template as TemplateName;
    68→  const map: Record<TemplateName, string> = {
    69→    modern:   `bg-gradient-to-br from-white via-white to-gray-50`,
    70→    classic:  `bg-gradient-to-br from-amber-100 via-amber-50 to-orange-50`,
    71→    luxury:   `bg-gradient-to-br from-gray-900 via-gray-800 to-gray-900`,
    72→    minimal:  `bg-white border-b border-gray-200`,
    73→    fastfood: `bg-gradient-to-br from-yellow-400 via-orange-400 to-red-400`,
    74→    cafe:     `bg-gradient-to-br from-amber-800 via-amber-700 to-yellow-800`,
    75→    pizza:    `bg-gradient-to-br from-green-700 via-green-600 to-red-700`,
    76→    dark:     `bg-gradient-to-br from-[#1a1a2e] via-[#16213e] to-[#0f3460]`,
    77→    colorful: `bg-gradient-to-r from-purple-500 via-pink-500 to-orange-400`,
    78→    elegant:  `bg-gradient-to-br from-stone-100 via-stone-50 to-stone-100`,
    79→  };
    80→  return map[t] ?? map.modern;
    81→}
    82→
    83→function getHeaderTextColor(template: string): string {
    84→  const dark = ['luxury', 'dark', 'fastfood', 'cafe', 'pizza', 'colorful'];
    85→  return dark.includes(template) ? 'text-white' : 'text-gray-900';
    86→}
    87→
    88→function getCategoryBtnStyle(template: string, primaryColor: string, isActive: boolean): React.CSSProperties {
    89→  const t = template as TemplateName;
    90→  if (isActive) {
    91→    if (t === 'luxury' || t === 'dark') return { backgroundColor: primaryColor, color: '#fff' };
    92→    if (t === 'colorful') return { background: `linear-gradient(135deg, ${primaryColor}, #ec4899)`, color: '#fff' };
    93→    return { backgroundColor: primaryColor, color: '#fff' };
    94→  }
    95→  if (t === 'luxury' || t === 'dark') return { backgroundColor: 'rgba(255,255,255,0.08)', color: 'rgba(255,255,255,0.6)' };
    96→  return { backgroundColor: '#f3f4f6', color: '#374151' };
    97→}

   137→

   154→

   156→  const addToBasket = (item: MenuItem) => {
   157→    setBasket(prev => {

   159→      if (existing) {
   160→        return prev.map(b => b.id === item.id ? { ...b, quantity: b.quantity + 1 } : b);
   161→      }

   165→  };
   166→
   167→  const updateBasketQty = (id: string, delta: number) => {

   169→      if (b.id !== id) return b;

   174→
   175→  const basketTotal = basket.reduce((sum, b) => sum + b.price * b.quantity, 0);
   176→  const basketCount = basket.reduce((sum, b) => sum + b.quantity, 0);

   179→    const phone = business.whatsapp?.replace(/[^0-9]/g, '');
   180→    if (!phone) return;
   181→

   183→    const itemsStr = basket.map(b => `- ${b.name} x${b.quantity} (${formatPrice(b.price * b.quantity)})`).join('\n');
   184→    const totalStr = `\n\nTotal: ${formatPrice(basketTotal)}`;
   185→    const message = encodeURIComponent(`${greeting}\n\n${itemsStr}${totalStr}`);
   186→    window.open(`https://wa.me/${phone}?text=${message}`, '_blank');
   187→  };
   188→
   189→  const primaryColor = business.primaryColor || '#10b981';
   190→  const secondaryColor = business.secondaryColor || '#059669';
   191→  const templateName = business.templateName || 'modern';
   192→  const fontMap: Record<string, React.CSSProperties['fontFamily']> = {
   193→    inter: 'system-ui, -apple-system, sans-serif',
   194→    serif: 'Georgia, "Times New Roman", serif',
   195→    mono: '"Courier New", monospace',
   196→    playfair: '"Playfair Display", Georgia, serif',
   197→  };
   198→  const fontFamily = fontMap[business.fontFamily] || fontMap.inter;
   199→
   200→  // Show cover overlay if not yet opened
   201→  const showCover = coverOpen && business.status === 'published';
   202→
   203→  /* ------------------------------------------------------------------ */

   205→  /* ------------------------------------------------------------------ */
   206→  if (showCover) {
   207→    return (
   208→      <>
   209→        <PublicCover business={business} onOpen={() => setCoverOpen(false)} />

   212→  }

   214→  if (business.status !== 'published') {

   216→      <div className="min-h-screen bg-gray-50 flex items-center justify-center px-4">
   217→        <div className="text-center max-w-sm">
   218→          <div className="w-16 h-16 rounded-2xl bg-amber-100 flex items-center justify-center mx-auto mb-4">

   221→          <h2 className="text-xl font-bold text-gray-900 mb-2">This Digital Experience is Currently Unavailable</h2>
   222→          <p className="text-gray-500 text-sm">
   223→            This business is updating their digital catalog. Please check back soon.
   224→          </p>

   226→      </div>
   227→    );
   228→  }
   229→
   230→  /* ------------------------------------------------------------------ */
   231→  /*  PUBLISHED UPLOAD DETECTION                                         */
   232→  /* ------------------------------------------------------------------ */
   233→  const publishedUploads = uploads.filter((u: Record<string, unknown>) => u.published === true);

   235→

   238→

   241→  /* ------------------------------------------------------------------ */
   242→  if (viewMode === 'flipbook' && business.flipbookEnabled) {

   258→      <div className="bg-gray-100" style={{ fontFamily }}>

   264→            style={{ backgroundColor: primaryColor, color: '#fff' }}

   280→                <p className="font-semibold text-sm">{basketCount} item{basketCount > 1 ? 's' : ''}</p>
   281→                <p className="text-xs text-muted-foreground">{formatPrice(basketTotal)}</p>

   291→              </div>
   292→            </div>
   293→          </div>
   294→        )}

   296→        <FlipbookMenu
   297→          business={business}
   298→          categories={categories}
   299→          items={items}
   300→          isPreview={false}
   301→        />
   302→      </div>
   303→    );
   304→  }
   305→
   306→  /* ------------------------------------------------------------------ */
   307→  /*  LIST MODE                                                          */
   308→  /* ------------------------------------------------------------------ */
   309→  const headerTextColor = getHeaderTextColor(templateName);
   310→
   311→  // If there's a published upload, show it in a simple viewer (non-flipbook list mode)
   312→  if (hasPublishedUpload && publishedUpload && !business.flipbookEnabled) {
   313→    const isImage = publishedUpload.fileType?.startsWith('image/');
   314→    return (
   315→      <div className={getTemplateClasses(templateName, primaryColor)} style={{ fontFamily }}>
   316→        {/* Header */}

   318→          <div className="max-w-2xl mx-auto text-center">
   319→            {business.logo ? (
   320→              <img src={business.logo} alt={business.name} className="w-20 h-20 rounded-2xl object-cover mx-auto mb-4 shadow-lg ring-4 ring-white/20" />
   321→            ) : (
   322→              <div className="w-20 h-20 rounded-2xl flex items-center justify-center mx-auto mb-4 text-2xl font-bold shadow-lg ring-4 ring-white/20"
   323→                style={{ backgroundColor: `${primaryColor}30`, color: headerTextColor === 'text-white' ? '#fff' : primaryColor }}>
   324→                {business.name.charAt(0).toUpperCase()}
   325→              </div>
   326→            )}
   327→            <h1 className="text-2xl font-bold mb-1">{business.name}</h1>
   328→            {business.description && (
   329→              <p className={`text-sm opacity-80 line-clamp-2 max-w-md mx-auto ${headerTextColor === 'text-white' ? 'text-white/70' : 'text-gray-500'}`}>
   330→                {business.description}
   331→              </p>
   332→            )}
   333→          </div>
   334→        </div>
   335→
   336→        {/* Uploaded Menu Content */}
   337→        <div className="max-w-2xl mx-auto p-4">
   338→          <h2 className="text-lg font-bold mb-3 text-gray-900">Our Menu</h2>
   339→          {isImage ? (
   340→            <img src={publishedUpload.url} alt={publishedUpload.fileName} className="w-full rounded-xl shadow-md" />
   341→          ) : (
   342→            <div className="rounded-xl border border-gray-200 p-8 text-center">
   343→              <p className="text-gray-500 mb-3">{publishedUpload.fileName}</p>
   344→              <a href={publishedUpload.url} target="_blank" rel="noopener noreferrer"
   345→                className="inline-flex items-center gap-2 px-4 py-2 rounded-lg text-white text-sm font-medium"
   346→                style={{ backgroundColor: primaryColor }}
   347→              >
   348→                View PDF Menu

   350→            </div>

   353→

   359→    );
   360→  }

   362→  return (

   364→      {/* View mode switcher */}
   365→      {business.flipbookEnabled && (
   366→        <div className="fixed top-3 right-3 z-50 flex gap-1 p-1 bg-white/90 backdrop-blur rounded-lg shadow-sm">
   367→          <button
   368→            onClick={() => setViewMode('flipbook')}
   369→            className="px-3 py-1.5 text-xs font-medium rounded-md bg-gray-100 text-gray-600 hover:bg-gray-200 transition-colors"
   370→          >
   371→            Flipbook
   372→          </button>
   373→          <button
   374→            onClick={() => setViewMode('list')}
   375→            className="px-3 py-1.5 text-xs font-medium rounded-md transition-colors"
   376→            style={{ backgroundColor: primaryColor, color: '#fff' }}
   377→          >
   378→            List
   379→          </button>
   380→        </div>

   383→      {/* Header */}
   384→      <div className={`${getHeaderBg(templateName, primaryColor)} ${headerTextColor} px-6 pt-8 pb-6`}>

   386→          {business.logo ? (
   387→            <img
   388→              src={business.logo}
   389→              alt={business.name}
   390→              className="w-20 h-20 rounded-2xl object-cover mx-auto mb-4 shadow-lg ring-4 ring-white/20"
   391→            />

   397→              {business.name.charAt(0).toUpperCase()}

   415→                <MapPin className="h-3 w-3" />

   582→
   583→      {/* WhatsApp Basket (list mode) */}
   584→      {business.whatsappOrder && basket.length > 0 && (
