    64→// ─── Become a Designer Form ────────────────────────────────────────
    65→function BecomeDesignerForm({ onSubmitted }: { onSubmitted: () => void }) {
    66→  const { token, user } = useAuthStore();
    67→  const [form, setForm] = useState({
    68→    fullName: '', displayName: '', username: '', email: user?.email || '',
    69→    bio: '', country: '', portfolioLink: '', profileImage: '',
    70→    specialties: [] as string[], socialLinks: {}, agreedTerms: false,
    71→  });
    72→  const [loading, setLoading] = useState(false);
    73→  const [appStatus, setAppStatus] = useState<'idle' | 'pending'>('idle');
    74→
    75→  const specialtyOptions = ['Book Cover Design', 'Menu Design', 'Branding', 'UI/UX', 'Print Design', 'Illustration'];
    76→
    77→  const update = (k: string, v: string | boolean | string[]) => setForm((p) => ({ ...p, [k]: v }));
    78→  const toggleSpec = (s: string) => {
    79→    setForm((p) => ({
    80→      ...p,
    81→      specialties: p.specialties.includes(s) ? p.specialties.filter((x) => x !== s) : [...p.specialties, s],
    82→    }));
    83→  };
    84→
    85→  const handleSubmit = async (e: React.FormEvent) => {
    86→    e.preventDefault();
    87→    if (!form.agreedTerms) { toast.error('You must agree to the terms'); return; }
    88→    setLoading(true);
    89→    try {
    90→      const res = await fetch('/api/marketplace/designers', {
    91→        method: 'POST',
    92→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
    93→        body: JSON.stringify(form),
    94→      });
    95→      const data = await res.json();
    96→      if (!res.ok) throw new Error(data.error || 'Failed');
    97→      toast.success('Application submitted!');
    98→      setAppStatus('pending');
    99→      onSubmitted();
   100→    } catch (err: any) {
   101→      toast.error(err.message);
   102→    } finally { setLoading(false); }
   103→  };
   104→
   105→  if (appStatus === 'pending') {
   106→    return (
   107→      <Card className="border-gold/20 bg-champagne/20">
   108→        <CardContent className="flex flex-col items-center justify-center py-16 text-center">
   109→          <div className="flex size-16 items-center justify-center rounded-full bg-gold/10">
   110→            <Clock className="size-8 text-gold" />
   111→          </div>
   112→          <h3 className="mt-4 text-xl font-bold text-charcoal">Application Pending Review</h3>
   113→          <p className="mt-2 max-w-md text-sm text-slate-600">
   114→            Your designer application is being reviewed. We'll notify you once it's approved.
   115→          </p>
   116→        </CardContent>
   117→      </Card>
   118→    );
   119→  }
   120→
   121→  return (
   122→    <Card className="border-gold/15 bg-gradient-to-br from-champagne/20 to-ivory">
   123→      <CardHeader>
   124→        <div className="flex items-center gap-3">
   125→          <div className="flex size-12 items-center justify-center rounded-xl bg-charcoal">
   126→            <PenTool className="size-6 text-gold" />
   127→          </div>
   128→          <div>
   129→            <CardTitle className="text-charcoal">Become a Designer</CardTitle>
   130→            <CardDescription>Share your design talent with the BizFlip community</CardDescription>
   131→          </div>
   132→        </div>
   133→      </CardHeader>
   134→      <CardContent>
   135→        <form onSubmit={handleSubmit} className="space-y-4">
   136→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   137→            <div><Label>Full Name *</Label><Input value={form.fullName} onChange={(e) => update('fullName', e.target.value)} required /></div>
   138→            <div><Label>Display Name *</Label><Input value={form.displayName} onChange={(e) => update('displayName', e.target.value)} required /></div>
   139→            <div><Label>Username *</Label><Input value={form.username} onChange={(e) => update('username', e.target.value)} required /></div>
   140→            <div><Label>Email *</Label><Input type="email" value={form.email} onChange={(e) => update('email', e.target.value)} required /></div>
   141→            <div className="sm:col-span-2"><Label>Country *</Label><Input value={form.country} onChange={(e) => update('country', e.target.value)} required /></div>
   142→          </div>
   143→          <div><Label>Bio *</Label><Textarea value={form.bio} onChange={(e) => update('bio', e.target.value)} rows={3} required /></div>
   144→          <div><Label>Portfolio Link</Label><Input placeholder="https://..." value={form.portfolioLink} onChange={(e) => update('portfolioLink', e.target.value)} /></div>
   145→          <div>
   146→            <Label className="mb-2 block">Design Specialties</Label>
   147→            <div className="flex flex-wrap gap-2">
   148→              {specialtyOptions.map((s) => (
   149→                <Badge key={s} variant={form.specialties.includes(s) ? 'default' : 'outline'}
   150→                  className={`cursor-pointer transition-colors ${form.specialties.includes(s) ? 'bg-charcoal text-white hover:bg-charcoal-light' : 'border-gold/20 text-slate-600 hover:bg-gold/10'}`}
   151→                  onClick={() => toggleSpec(s)}>{s}</Badge>
   152→              ))}
   153→            </div>
   154→          </div>
   155→          <div className="flex items-center gap-2">
   156→            <Switch checked={form.agreedTerms} onCheckedChange={(v) => update('agreedTerms', v)} id="terms" />
   157→            <Label htmlFor="terms" className="text-sm text-slate-600">I agree to the Designer Terms of Service</Label>
   158→          </div>
   159→          <Button type="submit" disabled={loading} className="w-full bg-charcoal hover:bg-charcoal-light text-white">
   160→            {loading && <Loader2 className="size-4 animate-spin" />}Submit Application
   161→          </Button>
   162→        </form>
   163→      </CardContent>
   164→    </Card>
   165→  );
   166→}

   168→// ─── Status Badge ─────────────────────────────────────────────────
   169→function StatusBadge({ status }: { status: string }) {
   170→  const map: Record<string, string> = {
   171→    published: 'bg-emerald-100 text-emerald-700', pending: 'bg-amber-100 text-amber-700',
   172→    rejected: 'bg-red-100 text-red-700', approved: 'bg-emerald-100 text-emerald-700',
   173→    available: 'bg-emerald-100 text-emerald-700', paid: 'bg-blue-100 text-blue-700',
   174→    reversed: 'bg-red-100 text-red-700', processing: 'bg-amber-100 text-amber-700',
   175→  };
   176→  return <Badge className={`${map[status] || 'bg-slate-100 text-slate-600'} text-xs`}>{status}</Badge>;
   177→}
