     1→'use client';
     2→
     3→import { useState, useEffect, useRef } from 'react';
     4→import {
     5→  Save,
     6→  Upload,
     7→  Trash2,
     8→  Loader2,
     9→  ImageIcon,
    10→  AlertTriangle,
    11→} from 'lucide-react';
    12→import {
    13→  Card,
    14→  CardContent,
    15→  CardHeader,
    16→  CardTitle,
    17→  CardDescription,
    18→} from '@/components/ui/card';
    19→import { Button } from '@/components/ui/button';
    20→import { Input } from '@/components/ui/input';
    21→import { Textarea } from '@/components/ui/textarea';
    22→import { Label } from '@/components/ui/label';
    23→import { Switch } from '@/components/ui/switch';
    24→import { Separator } from '@/components/ui/separator';
    25→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    26→import {
    27→  Select,
    28→  SelectContent,
    29→  SelectItem,
    30→  SelectTrigger,
    31→  SelectValue,
    32→} from '@/components/ui/select';
    33→import {
    34→  AlertDialog,
    35→  AlertDialogAction,
    36→  AlertDialogCancel,
    37→  AlertDialogContent,
    38→  AlertDialogDescription,
    39→  AlertDialogFooter,
    40→  AlertDialogHeader,
    41→  AlertDialogTitle,
    42→  AlertDialogTrigger,
    43→} from '@/components/ui/alert-dialog';
    44→import { useAuthStore, useAppStore } from '@/lib/store';
    45→import type { Business } from '@/lib/types';
    46→import { BUSINESS_CATEGORIES } from '@/lib/types';
    47→import { toast } from 'sonner';
    48→
    49→export default function SettingsPanel() {
    50→  const { token, user } = useAuthStore();
    51→  const { currentBusiness, setCurrentBusiness } = useAppStore();
    52→  const fileInputRef = useRef<HTMLInputElement>(null);
    53→
    54→  // Business form state
    55→  const [bizForm, setBizForm] = useState({
    56→    name: '',
    57→    category: '',
    58→    phone: '',
    59→    whatsapp: '',
    60→    address: '',
    61→    openingHours: '',
    62→    description: '',
    63→    status: 'draft' as string,
    64→    whatsappOrder: false,
    65→    seoEnabled: false,
    66→  });
    67→  const [bizLoading, setBizLoading] = useState(false);
    68→  const [bizSaving, setBizSaving] = useState(false);
    69→  const [bizInitial, setBizInitial] = useState<Business | null>(null);
    70→
    71→  // Account form state
    72→  const [accForm, setAccForm] = useState({
    73→    name: '',
    74→    phone: '',
    75→    currentPassword: '',
    76→    newPassword: '',
    77→    confirmPassword: '',
    78→  });
    79→  const [accSaving, setAccSaving] = useState(false);
    80→  const [accLoading, setAccLoading] = useState(true);
    81→
    82→  // Upload state
    83→  const [uploading, setUploading] = useState(false);
    84→
    85→  // Fetch business data
    86→  useEffect(() => {
    87→    if (!currentBusiness?.id || !token) return;
    88→    setBizLoading(true);
    89→    async function fetchBusiness() {
    90→      try {
    91→        const res = await fetch(`/api/businesses/${currentBusiness.id}`, {
    92→          headers: { Authorization: `Bearer ${token}` },
    93→        });
    94→        if (res.ok) {
    95→          const data = await res.json();
    96→          const b: Business = data.business;
    97→          setBizInitial(b);
    98→          setBizForm({
    99→            name: b.name || '',
   100→            category: b.category || '',
   101→            phone: b.phone || '',
   102→            whatsapp: b.whatsapp || '',
   103→            address: b.address || '',
   104→            openingHours: b.openingHours || '',
   105→            description: b.description || '',
   106→            status: b.status || 'draft',
   107→            whatsappOrder: b.whatsappOrder ?? false,
   108→            seoEnabled: b.seoEnabled ?? false,
   109→          });
   110→        }
   111→      } catch {
   112→        toast.error('Failed to load business settings');
   113→      } finally {
   114→        setBizLoading(false);
   115→      }
   116→    }
   117→    fetchBusiness();
   118→  }, [currentBusiness?.id, token]);
   119→
   120→  // Fetch user data
   121→  useEffect(() => {
   122→    if (!user || !token) return;
   123→    setAccForm({
   124→      name: user.name || '',
   125→      phone: '',
   126→      currentPassword: '',
   127→      newPassword: '',
   128→      confirmPassword: '',
   129→    });
   130→    setAccLoading(false);
   131→  }, [user, token]);
   132→
   133→  // Save business settings
   134→  const handleSaveBusiness = async () => {
   135→    if (!currentBusiness?.id || !token) return;
   136→    setBizSaving(true);
   137→    try {
   138→      const res = await fetch(`/api/businesses/${currentBusiness.id}`, {
   139→        method: 'PUT',
   140→        headers: {
   141→          Authorization: `Bearer ${token}`,
   142→          'Content-Type': 'application/json',
   143→        },
   144→        body: JSON.stringify(bizForm),
   145→      });
   146→      if (res.ok) {
   147→        const data = await res.json();
   148→        const updated: Business = data.business;
   149→        setBizInitial(updated);
   150→        setCurrentBusiness({
   151→          id: updated.id,
   152→          slug: updated.slug,
   153→          name: updated.name,
   154→          logo: updated.logo,
   155→          status: updated.status,
   156→        });
   157→        toast.success('Business settings saved');
   158→      } else {
   159→        const err = await res.json().catch(() => ({}));
   160→        toast.error(err.error || 'Failed to save settings');
   161→      }
   162→    } catch {
   163→      toast.error('Failed to save settings');
   164→    } finally {
   165→      setBizSaving(false);
   166→    }
   167→  };
   168→
   169→  // Save account settings
   170→  const handleSaveAccount = async () => {
   171→    if (!token) return;
   172→    const updateData: Record<string, string> = {};
   173→    if (accForm.name) updateData.name = accForm.name;
   174→    if (accForm.phone) updateData.phone = accForm.phone;
   175→
   176→    // Password change validation
   177→    if (accForm.newPassword) {
   178→      if (!accForm.currentPassword) {
   179→        toast.error('Current password is required to change password');
   180→        return;
   181→      }
   182→      if (accForm.newPassword !== accForm.confirmPassword) {
   183→        toast.error('New passwords do not match');
   184→        return;
   185→      }
   186→      if (accForm.newPassword.length < 8) {
   187→        toast.error('Password must be at least 8 characters');
   188→        return;
   189→      }
   190→      updateData.currentPassword = accForm.currentPassword;
   191→      updateData.newPassword = accForm.newPassword;
   192→    }
   193→
   194→    if (Object.keys(updateData).length === 0) {
   195→      toast.error('No changes to save');
   196→      return;
   197→    }
   198→
   199→    setAccSaving(true);
   200→    try {
   201→      const res = await fetch('/api/auth/me', {
   202→        method: 'PUT',
   203→        headers: {
   204→          Authorization: `Bearer ${token}`,
   205→          'Content-Type': 'application/json',
   206→        },
   207→        body: JSON.stringify(updateData),
   208→      });
   209→      if (res.ok) {
   210→        toast.success('Account settings saved');
   211→        setAccForm((prev) => ({
   212→          ...prev,
   213→          currentPassword: '',
   214→          newPassword: '',
   215→          confirmPassword: '',
   216→        }));
   217→      } else {
   218→        const err = await res.json().catch(() => ({}));
   219→        toast.error(err.error || 'Failed to save account');
   220→      }
   221→    } catch {
   222→      toast.error('Failed to save account');
   223→    } finally {
   224→      setAccSaving(false);
   225→    }
   226→  };
   227→
   228→  // Logo upload
   229→  const handleLogoUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   230→    const file = e.target.files?.[0];
   231→    if (!file || !currentBusiness?.id || !token) return;
   232→    setUploading(true);
   233→    try {
   234→      const formData = new FormData();
   235→      formData.append('file', file);
   236→      formData.append('businessId', currentBusiness.id);
   237→      formData.append('type', 'logo');
   238→      const res = await fetch('/api/upload', {
   239→        method: 'POST',
   240→        headers: { Authorization: `Bearer ${token}` },
   241→        body: formData,
   242→      });
   243→      if (res.ok) {
   244→        const data = await res.json();
   245→        if (data.url) {
   246→          setCurrentBusiness({
   247→            ...currentBusiness,
   248→            logo: data.url,
   249→          });
   250→          toast.success('Logo uploaded');
   251→        }
   252→      } else {
   253→        toast.error('Failed to upload logo');
   254→      }
   255→    } catch {
   256→      toast.error('Failed to upload logo');
   257→    } finally {
   258→      setUploading(false);
   259→      if (fileInputRef.current) fileInputRef.current.value = '';
   260→    }
   261→  };
   262→
   263→  // Delete account
   264→  const handleDeleteAccount = async () => {
   265→    if (!token) return;
   266→    try {
   267→      const res = await fetch('/api/auth/me', {
   268→        method: 'DELETE',
   269→        headers: { Authorization: `Bearer ${token}` },
   270→      });
   271→      if (res.ok) {
   272→        toast.success('Account deleted');
   273→        useAuthStore.getState().logout();
   274→      } else {
   275→        toast.error('Failed to delete account');
   276→      }
   277→    } catch {
   278→      toast.error('Failed to delete account');
   279→    }
   280→  };
   281→
   282→  if (!currentBusiness) {
   283→    return (
   284→      <div className="flex flex-col items-center justify-center py-20 gap-3 text-muted-foreground">
   285→        <p>Select a business to manage settings</p>
   286→      </div>
   287→    );
   288→  }
   289→
   290→  const hasBizChanges =
   291→    bizInitial &&
   292→    (bizForm.name !== bizInitial.name ||
   293→      bizForm.category !== (bizInitial.category || '') ||
   294→      bizForm.phone !== (bizInitial.phone || '') ||
   295→      bizForm.whatsapp !== (bizInitial.whatsapp || '') ||
   296→      bizForm.address !== (bizInitial.address || '') ||
   297→      bizForm.openingHours !== (bizInitial.openingHours || '') ||
   298→      bizForm.description !== (bizInitial.description || '') ||
   299→      bizForm.status !== bizInitial.status ||
   300→      bizForm.whatsappOrder !== (bizInitial.whatsappOrder ?? false) ||
   301→      bizForm.seoEnabled !== (bizInitial.seoEnabled ?? false));
   302→
   303→  return (
   304→    <Tabs defaultValue="business" className="space-y-6">
   305→      <TabsList>
   306→        <TabsTrigger value="business">Business Settings</TabsTrigger>
   307→        <TabsTrigger value="account">Account Settings</TabsTrigger>
   308→      </TabsList>
   309→
   310→      {/* Business Settings Tab */}
   311→      <TabsContent value="business" className="space-y-6">
   312→        {/* Basic Info */}
   313→        <Card>
   314→          <CardHeader>
   315→            <CardTitle>Basic Information</CardTitle>
   316→            <CardDescription>
   317→              Update your business details and branding
   318→            </CardDescription>
   319→          </CardHeader>
   320→          <CardContent className="space-y-4">
   321→            {bizLoading ? (
   322→              <div className="space-y-4">
   323→                {[1, 2, 3, 4].map((i) => (
   324→                  <div key={i} className="space-y-2">
   325→                    <div className="h-4 w-24 rounded bg-muted" />
   326→                    <div className="h-9 rounded-md bg-muted" />
   327→                  </div>
   328→                ))}
   329→              </div>
   330→            ) : (
   331→              <>
   332→                {/* Logo Upload */}
   333→                <div className="space-y-2">
   334→                  <Label>Logo</Label>
   335→                  <div className="flex items-center gap-4">
   336→                    {currentBusiness.logo ? (
   337→                      <img
   338→                        src={currentBusiness.logo}
   339→                        alt="Logo"
   340→                        className="h-16 w-16 rounded-xl object-cover ring-1 ring-border"
   341→                      />
   342→                    ) : (
   343→                      <div className="flex h-16 w-16 items-center justify-center rounded-xl bg-muted">
   344→                        <ImageIcon className="h-6 w-6 text-muted-foreground" />
   345→                      </div>
   346→                    )}
   347→                    <div className="flex flex-col gap-2">
   348→                      <Button
   349→                        type="button"
   350→                        variant="outline"
   351→                        size="sm"
   352→                        onClick={() => fileInputRef.current?.click()}
   353→                        disabled={uploading}
   354→                      >
   355→                        {uploading ? (
   356→                          <Loader2 className="mr-1.5 h-4 w-4 animate-spin" />
   357→                        ) : (
   358→                          <Upload className="mr-1.5 h-4 w-4" />
   359→                        )}
   360→                        {uploading ? 'Uploading...' : 'Upload Logo'}
   361→                      </Button>
   362→                      <input
   363→                        ref={fileInputRef}
   364→                        type="file"
   365→                        accept="image/*"
   366→                        onChange={handleLogoUpload}
   367→                        className="hidden"
   368→                      />
   369→                      <p className="text-xs text-muted-foreground">
   370→                        PNG, JPG up to 2MB
   371→                      </p>
   372→                    </div>
   373→                  </div>
   374→                </div>
   375→
   376→                <Separator />
   377→
   378→                {/* Business Name */}
   379→                <div className="space-y-2">
   380→                  <Label htmlFor="biz-name">Business Name</Label>
   381→                  <Input
   382→                    id="biz-name"
   383→                    value={bizForm.name}
   384→                    onChange={(e) =>
   385→                      setBizForm((p) => ({ ...p, name: e.target.value }))
   386→                    }
   387→                    placeholder="Enter business name"
   388→                  />
   389→                </div>
   390→
   391→                {/* Category */}
   392→                <div className="space-y-2">
   393→                  <Label>Category</Label>
   394→                  <Select
   395→                    value={bizForm.category}
   396→                    onValueChange={(v) =>
   397→                      setBizForm((p) => ({ ...p, category: v }))
   398→                    }
   399→                  >
   400→                    <SelectTrigger className="w-full">
   401→                      <SelectValue placeholder="Select a category" />
   402→                    </SelectTrigger>
   403→                    <SelectContent>
   404→                      {BUSINESS_CATEGORIES.map((cat) => (
   405→                        <SelectItem key={cat} value={cat}>
   406→                          {cat}
   407→                        </SelectItem>
   408→                      ))}
   409→                    </SelectContent>
   410→                  </Select>
   411→                </div>
   412→
   413→                {/* Phone & WhatsApp */}
   414→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   415→                  <div className="space-y-2">
   416→                    <Label htmlFor="biz-phone">Phone</Label>
   417→                    <Input
   418→                      id="biz-phone"
   419→                      value={bizForm.phone}
   420→                      onChange={(e) =>
   421→                        setBizForm((p) => ({ ...p, phone: e.target.value }))
   422→                      }
   423→                      placeholder="Phone number"
   424→                    />
   425→                  </div>
   426→                  <div className="space-y-2">
   427→                    <Label htmlFor="biz-whatsapp">WhatsApp</Label>
   428→                    <Input
   429→                      id="biz-whatsapp"
   430→                      value={bizForm.whatsapp}
   431→                      onChange={(e) =>
   432→                        setBizForm((p) => ({ ...p, whatsapp: e.target.value }))
   433→                      }
   434→                      placeholder="WhatsApp number"
   435→                    />
   436→                  </div>
   437→                </div>
   438→
   439→                {/* Address */}
   440→                <div className="space-y-2">
   441→                  <Label htmlFor="biz-address">Address</Label>
   442→                  <Input
   443→                    id="biz-address"
   444→                    value={bizForm.address}
   445→                    onChange={(e) =>
   446→                      setBizForm((p) => ({ ...p, address: e.target.value }))
   447→                    }
   448→                    placeholder="Business address"
   449→                  />
   450→                </div>
   451→
   452→                {/* Opening Hours */}
   453→                <div className="space-y-2">
   454→                  <Label htmlFor="biz-hours">Opening Hours</Label>
   455→                  <Input
   456→                    id="biz-hours"
   457→                    value={bizForm.openingHours}
   458→                    onChange={(e) =>
   459→                      setBizForm((p) => ({ ...p, openingHours: e.target.value }))
   460→                    }
   461→                    placeholder="e.g. Mon-Fri 9AM-10PM, Sat-Sun 10AM-11PM"
   462→                  />
   463→                </div>
   464→
   465→                {/* Description */}
   466→                <div className="space-y-2">
   467→                  <Label htmlFor="biz-desc">Description</Label>
   468→                  <Textarea
   469→                    id="biz-desc"
   470→                    value={bizForm.description}
   471→                    onChange={(e) =>
   472→                      setBizForm((p) => ({ ...p, description: e.target.value }))
   473→                    }
   474→                    placeholder="Brief description of your business"
   475→                    rows={3}
   476→                  />
   477→                </div>
   478→              </>
   479→            )}
   480→          </CardContent>
   481→        </Card>
   482→
   483→        {/* Menu Settings */}
   484→        <Card>
   485→          <CardHeader>
   486→            <CardTitle>Menu Settings</CardTitle>
   487→            <CardDescription>
   488→              Control how your menu is published and accessed
   489→            </CardDescription>
   490→          </CardHeader>
   491→          <CardContent className="space-y-6">
   492→            {bizLoading ? (
   493→              <div className="space-y-4">
   494→                {[1, 2, 3].map((i) => (
   495→                  <div key={i} className="flex items-center justify-between">
   496→                    <div className="space-y-1">
   497→                      <div className="h-4 w-32 rounded bg-muted" />
   498→                      <div className="h-3 w-48 rounded bg-muted" />
   499→                    </div>
   500→                    <div className="h-5 w-10 rounded-full bg-muted" />
   501→                  </div>
   502→                ))}
   503→              </div>
   504→            ) : (
   505→              <>
   506→                {/* Menu Status */}
   507→                <div className="space-y-2">
   508→                  <Label>Menu Status</Label>
   509→                  <Select
   510→                    value={bizForm.status}
   511→                    onValueChange={(v) =>
   512→                      setBizForm((p) => ({ ...p, status: v }))
   513→                    }
   514→                  >
   515→                    <SelectTrigger className="w-full sm:w-64">
   516→                      <SelectValue />
   517→                    </SelectTrigger>
   518→                    <SelectContent>
   519→                      <SelectItem value="draft">Draft</SelectItem>
   520→                      <SelectItem value="published">Published</SelectItem>
   521→                      <SelectItem value="unpublished">Unpublished</SelectItem>
   522→                    </SelectContent>
   523→                  </Select>
   524→                  <p className="text-xs text-muted-foreground">
   525→                    Only published menus are accessible via QR code
   526→                  </p>
   527→                </div>
   528→
   529→                <Separator />
   530→
   531→                {/* WhatsApp Ordering */}
   532→                <div className="flex items-center justify-between gap-4">
   533→                  <div className="space-y-0.5">
   534→                    <Label>WhatsApp Ordering</Label>
   535→                    <p className="text-sm text-muted-foreground">
   536→                      Allow customers to place orders via WhatsApp
   537→                    </p>
   538→                  </div>
   539→                  <Switch
   540→                    checked={bizForm.whatsappOrder}
   541→                    onCheckedChange={(v) =>
   542→                      setBizForm((p) => ({ ...p, whatsappOrder: v }))
   543→                    }
   544→                  />
   545→                </div>
   546→
   547→                <Separator />
   548→
   549→                {/* SEO */}
   550→                <div className="flex items-center justify-between gap-4">
   551→                  <div className="space-y-0.5">
   552→                    <Label>SEO Indexing</Label>
   553→                    <p className="text-sm text-muted-foreground">
   554→                      Allow search engines to index your menu page
   555→                    </p>
   556→                  </div>
   557→                  <Switch
   558→                    checked={bizForm.seoEnabled}
   559→                    onCheckedChange={(v) =>
   560→                      setBizForm((p) => ({ ...p, seoEnabled: v }))
   561→                    }
   562→                  />
   563→                </div>
   564→              </>
   565→            )}
   566→          </CardContent>
   567→        </Card>
   568→
   569→        {/* Save Button */}
   570→        <div className="flex justify-end">
   571→          <Button
   572→            onClick={handleSaveBusiness}
   573→            disabled={bizSaving || bizLoading || !hasBizChanges}
   574→            className="bg-emerald-600 hover:bg-emerald-700"
   575→          >
   576→            {bizSaving ? (
   577→              <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   578→            ) : (
   579→              <Save className="mr-2 h-4 w-4" />
   580→            )}
   581→            Save Changes
   582→          </Button>
   583→        </div>
   584→      </TabsContent>
   585→
   586→      {/* Account Settings Tab */}
   587→      <TabsContent value="account" className="space-y-6">
   588→        {/* Profile Info */}
   589→        <Card>
   590→          <CardHeader>
   591→            <CardTitle>Profile Information</CardTitle>
   592→            <CardDescription>Update your personal details</CardDescription>
   593→          </CardHeader>
   594→          <CardContent className="space-y-4">
   595→            {accLoading ? (
   596→              <div className="space-y-4">
   597→                {[1, 2, 3].map((i) => (
   598→                  <div key={i} className="space-y-2">
   599→                    <div className="h-4 w-20 rounded bg-muted" />
   600→                    <div className="h-9 rounded-md bg-muted" />
   601→                  </div>
   602→                ))}
   603→              </div>
   604→            ) : (
   605→              <>
   606→                <div className="space-y-2">
   607→                  <Label htmlFor="acc-email">Email</Label>
   608→                  <Input
   609→                    id="acc-email"
   610→                    value={user?.email || ''}
   611→                    disabled
   612→                    className="bg-muted"
   613→                  />
   614→                  <p className="text-xs text-muted-foreground">
   615→                    Email cannot be changed
   616→                  </p>
   617→                </div>
   618→
   619→                <div className="space-y-2">
   620→                  <Label htmlFor="acc-name">Full Name</Label>
   621→                  <Input
   622→                    id="acc-name"
   623→                    value={accForm.name}
   624→                    onChange={(e) =>
   625→                      setAccForm((p) => ({ ...p, name: e.target.value }))
   626→                    }
   627→                    placeholder="Your full name"
   628→                  />
   629→                </div>
   630→
   631→                <div className="space-y-2">
   632→                  <Label htmlFor="acc-phone">Phone</Label>
   633→                  <Input
   634→                    id="acc-phone"
   635→                    value={accForm.phone}
   636→                    onChange={(e) =>
   637→                      setAccForm((p) => ({ ...p, phone: e.target.value }))
   638→                    }
   639→                    placeholder="Your phone number"
   640→                  />
   641→                </div>
   642→              </>
   643→            )}
   644→          </CardContent>
