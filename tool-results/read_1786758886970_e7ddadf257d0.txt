    44→    44→type TemplateName =
    45→    45→  | 'modern' | 'classic' | 'luxury' | 'minimal' | 'fastfood'

    47→    47→
    48→    48→function getTemplateClasses(template: string, primaryColor: string): string {
    49→    49→  const t = template as TemplateName;
    50→    50→  const base = 'min-h-screen';
    51→    51→  const overrides: Record<TemplateName, string> = {
    52→    52→    modern: 'bg-white text-gray-900',
    53→    53→    classic: 'bg-amber-50/50 text-gray-900 font-serif',
    54→    54→    luxury: 'bg-gray-950 text-gray-100',
    55→    55→    minimal: 'bg-white text-gray-900',
    56→    56→    fastfood: 'bg-yellow-50 text-gray-900',
    57→    57→    cafe: 'bg-orange-50/40 text-gray-900',
    58→    58→    pizza: 'bg-red-50/30 text-gray-900',
    59→    59→    dark: 'bg-[#1a1a2e] text-gray-100',
    60→    60→    colorful: 'bg-white text-gray-900',
    61→    61→    elegant: 'bg-stone-50 text-gray-800',
    62→    62→  };
    63→    63→  return `${base} ${overrides[t] ?? overrides.modern}`;
    64→    64→}
    65→    65→
    66→    66→function getHeaderBg(template: string, primaryColor: string): string {
    67→    67→  const t = template as TemplateName;
    68→    68→  const map: Record<TemplateName, string> = {
    69→    69→    modern:   `bg-gradient-to-br from-white via-white to-gray-50`,
    70→    70→    classic:  `bg-gradient-to-br from-amber-100 via-amber-50 to-orange-50`,
    71→    71→    luxury:   `bg-gradient-to-br from-gray-900 via-gray-800 to-gray-900`,
    72→    72→    minimal:  `bg-white border-b border-gray-200`,
    73→    73→    fastfood: `bg-gradient-to-br from-yellow-400 via-orange-400 to-red-400`,
    74→    74→    cafe:     `bg-gradient-to-br from-amber-800 via-amber-700 to-yellow-800`,
    75→    75→    pizza:    `bg-gradient-to-br from-green-700 via-green-600 to-red-700`,
    76→    76→    dark:     `bg-gradient-to-br from-[#1a1a2e] via-[#16213e] to-[#0f3460]`,
    77→    77→    colorful: `bg-gradient-to-r from-purple-500 via-pink-500 to-orange-400`,
    78→    78→    elegant:  `bg-gradient-to-br from-stone-100 via-stone-50 to-stone-100`,
    79→    79→  };
    80→    80→  return map[t] ?? map.modern;
    81→    81→}

    83→    83→function getHeaderTextColor(template: string): string {
    84→    84→  const dark = ['luxury', 'dark', 'fastfood', 'cafe', 'pizza', 'colorful'];
    85→    85→  return dark.includes(template) ? 'text-white' : 'text-gray-900';

    88→    88→function getCategoryBtnStyle(template: string, primaryColor: string, isActive: boolean): React.CSSProperties {
    89→    89→  const t = template as TemplateName;
    90→    90→  if (isActive) {
    91→    91→    if (t === 'luxury' || t === 'dark') return { backgroundColor: primaryColor, color: '#fff' };
    92→    92→    if (t === 'colorful') return { background: `linear-gradient(135deg, ${primaryColor}, #ec4899)`, color: '#fff' };
    93→    93→    return { backgroundColor: primaryColor, color: '#fff' };
    94→    94→  }
    95→    95→  if (t === 'luxury' || t === 'dark') return { backgroundColor: 'rgba(255,255,255,0.08)', color: 'rgba(255,255,255,0.6)' };
    96→    96→  return { backgroundColor: '#f3f4f6', color: '#374151' };
    97→    97→}
