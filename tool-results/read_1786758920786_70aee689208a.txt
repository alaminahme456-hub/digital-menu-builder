     1→'use client';
     2→
     3→import { useState, useEffect, useRef } from 'react';
     4→import {
     5→  Save,
     6→  Upload,
     7→  Trash2,
     8→  Loader2,
     9→  ImageIcon,
    10→  AlertTriangle,
    11→} from 'lucide-react';
    12→import {
    13→  Card,
    14→  CardContent,
    15→  CardHeader,
    16→  CardTitle,
    17→  CardDescription,
    18→} from '@/components/ui/card';
    19→import { Button } from '@/components/ui/button';
    20→import { Input } from '@/components/ui/input';
    21→import { Textarea } from '@/components/ui/textarea';
    22→import { Label } from '@/components/ui/label';
    23→import { Switch } from '@/components/ui/switch';
    24→import { Separator } from '@/components/ui/separator';
    25→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    26→import {
    27→  Select,
    28→  SelectContent,
    29→  SelectItem,
    30→  SelectTrigger,
    31→  SelectValue,
    32→} from '@/components/ui/select';
    33→import {
    34→  AlertDialog,
    35→  AlertDialogAction,
    36→  AlertDialogCancel,
    37→  AlertDialogContent,
    38→  AlertDialogDescription,
    39→  AlertDialogFooter,
    40→  AlertDialogHeader,
    41→  AlertDialogTitle,
    42→  AlertDialogTrigger,
    43→} from '@/components/ui/alert-dialog';
    44→import { useAuthStore, useAppStore } from '@/lib/store';
    45→import type { Business } from '@/lib/types';
    46→import { BUSINESS_CATEGORIES } from '@/lib/types';
    47→import { getAppUrl } from '@/lib/auth';
    48→import { toast } from 'sonner';
    49→
    50→interface SettingsPanelProps {
    51→  initialTab?: string;
    52→}
    53→
    54→export default function SettingsPanel({ initialTab = 'business' }: SettingsPanelProps) {
    55→  const { token, user } = useAuthStore();
    56→  const { currentBusiness, setCurrentBusiness } = useAppStore();
    57→  const fileInputRef = useRef<HTMLInputElement>(null);
    58→
    59→  // Business form state
    60→  const [bizForm, setBizForm] = useState({
    61→    name: '',
    62→    category: '',
    63→    phone: '',
    64→    whatsapp: '',
    65→    address: '',
    66→    openingHours: '',
    67→    description: '',
    68→    status: 'draft' as string,
    69→    whatsappOrder: false,
    70→    seoEnabled: false,
    71→  });
    72→  const [bizLoading, setBizLoading] = useState(false);
    73→  const [bizSaving, setBizSaving] = useState(false);
    74→  const [bizInitial, setBizInitial] = useState<Business | null>(null);
    75→
    76→  // Account form state
    77→  const [accForm, setAccForm] = useState({
    78→    name: '',
    79→    phone: '',
    80→    currentPassword: '',
    81→    newPassword: '',
    82→    confirmPassword: '',
    83→  });
    84→  const [accSaving, setAccSaving] = useState(false);
    85→  const [accLoading, setAccLoading] = useState(true);
    86→
    87→  // Upload state
    88→  const [uploading, setUploading] = useState(false);
    89→
    90→  // Flipbook settings state
    91→  const [flipbookForm, setFlipbookForm] = useState({
    92→    flipbookEnabled: true,
    93→    flipbookAnimEnabled: true,
    94→    flipbookAnimSpeed: 'medium' as string,
    95→    flipbookPageNumbers: true,
    96→    flipbookSwipeNav: true,
    97→    flipbookSoundEffects: false,
    98→    flipbookFullscreen: true,
    99→    flipbookInteractions: true,
   100→  });
   101→  const [flipbookSaving, setFlipbookSaving] = useState(false);
   102→
   103→  // Ordering settings state
   104→  const [orderingForm, setOrderingForm] = useState({
   105→    whatsappOrder: true,
   106→    basketEnabled: true,
   107→    showQuantitySelector: true,
   108→    showOrderButton: true,
   109→    whatsappGreeting: 'Hello, I would like to place an order:',
   110→  });
   111→  const [orderingSaving, setOrderingSaving] = useState(false);
   112→
   113→  // Fetch business data
   114→  useEffect(() => {
   115→    if (!currentBusiness?.id || !token) return;
   116→    setBizLoading(true);
   117→    async function fetchBusiness() {
   118→      try {
   119→        const res = await fetch(`/api/businesses/${currentBusiness.id}`, {
   120→          headers: { Authorization: `Bearer ${token}` },
   121→        });
   122→        if (res.ok) {
   123→          const data = await res.json();
   124→          const b: Business = data.business;
   125→          setBizInitial(b);
   126→          setBizForm({
   127→            name: b.name || '',
   128→            category: b.category || '',
   129→            phone: b.phone || '',
   130→            whatsapp: b.whatsapp || '',
   131→            address: b.address || '',
   132→            openingHours: b.openingHours || '',
   133→            description: b.description || '',
   134→            status: b.status || 'draft',
   135→            whatsappOrder: b.whatsappOrder ?? false,
   136→            seoEnabled: b.seoEnabled ?? false,
   137→            welcomeMessage: (b as Record<string, unknown>).welcomeMessage as string || '',
   138→          });
   139→          // Load flipbook settings
   140→          setFlipbookForm({
   141→            flipbookEnabled: b.flipbookEnabled ?? true,
   142→            flipbookAnimEnabled: b.flipbookAnimEnabled ?? true,
   143→            flipbookAnimSpeed: b.flipbookAnimSpeed || 'medium',
   144→            flipbookPageNumbers: b.flipbookPageNumbers ?? true,
   145→            flipbookSwipeNav: b.flipbookSwipeNav ?? true,
   146→            flipbookSoundEffects: b.flipbookSoundEffects ?? false,
   147→            flipbookFullscreen: b.flipbookFullscreen ?? true,
   148→            flipbookInteractions: b.flipbookInteractions ?? true,
   149→          });
   150→          // Load ordering settings
   151→          setOrderingForm({
   152→            whatsappOrder: b.whatsappOrder ?? true,
   153→            basketEnabled: b.basketEnabled ?? true,
   154→            showQuantitySelector: b.showQuantitySelector ?? true,
   155→            showOrderButton: b.showOrderButton ?? true,
   156→            whatsappGreeting: b.whatsappGreeting || 'Hello, I would like to place an order:',
   157→          });
   158→        }
   159→      } catch {
   160→        toast.error('Failed to load business settings');
   161→      } finally {
   162→        setBizLoading(false);
   163→      }
   164→    }
   165→    fetchBusiness();
   166→  }, [currentBusiness?.id, token]);
   167→
   168→  // Fetch user data
   169→  useEffect(() => {
   170→    if (!user || !token) return;
   171→    setAccForm({
   172→      name: user.name || '',
   173→      phone: '',
   174→      currentPassword: '',
   175→      newPassword: '',
   176→      confirmPassword: '',
   177→    });
   178→    setAccLoading(false);
   179→  }, [user, token]);
   180→
   181→  // Save business settings
   182→  const handleSaveBusiness = async () => {
   183→    if (!currentBusiness?.id || !token) return;
   184→    setBizSaving(true);
   185→    try {
   186→      const res = await fetch(`/api/businesses/${currentBusiness.id}`, {
   187→        method: 'PUT',
   188→        headers: {
   189→          Authorization: `Bearer ${token}`,
   190→          'Content-Type': 'application/json',
   191→        },
   192→        body: JSON.stringify(bizForm),
   193→      });
   194→      if (res.ok) {
   195→        const data = await res.json();
   196→        const updated: Business = data.business;
   197→        setBizInitial(updated);
   198→        setCurrentBusiness({
   199→          id: updated.id,
   200→          slug: updated.slug,
   201→          name: updated.name,
   202→          logo: updated.logo,
   203→          status: updated.status,
   204→        });
   205→        toast.success('Business settings saved');
   206→      } else {
   207→        const err = await res.json().catch(() => ({}));
   208→        toast.error(err.error || 'Failed to save settings');
   209→      }
   210→    } catch {
   211→      toast.error('Failed to save settings');
   212→    } finally {
   213→      setBizSaving(false);
   214→    }
   215→  };
   216→
   217→  // Save account settings
   218→  const handleSaveAccount = async () => {
   219→    if (!token) return;
   220→    const updateData: Record<string, string> = {};
   221→    if (accForm.name) updateData.name = accForm.name;
   222→    if (accForm.phone) updateData.phone = accForm.phone;
   223→
   224→    // Password change validation
   225→    if (accForm.newPassword) {
   226→      if (!accForm.currentPassword) {
   227→        toast.error('Current password is required to change password');
   228→        return;
   229→      }
   230→      if (accForm.newPassword !== accForm.confirmPassword) {
   231→        toast.error('New passwords do not match');
   232→        return;
   233→      }
   234→      if (accForm.newPassword.length < 8) {
   235→        toast.error('Password must be at least 8 characters');
   236→        return;
   237→      }
   238→      updateData.currentPassword = accForm.currentPassword;
   239→      updateData.newPassword = accForm.newPassword;
   240→    }
   241→
   242→    if (Object.keys(updateData).length === 0) {
   243→      toast.error('No changes to save');
   244→      return;
   245→    }
   246→
   247→    setAccSaving(true);
   248→    try {
   249→      const res = await fetch('/api/auth/me', {
   250→        method: 'PUT',
   251→        headers: {
   252→          Authorization: `Bearer ${token}`,
   253→          'Content-Type': 'application/json',
   254→        },
   255→        body: JSON.stringify(updateData),
   256→      });
   257→      if (res.ok) {
   258→        toast.success('Account settings saved');
   259→        setAccForm((prev) => ({
   260→          ...prev,
   261→          currentPassword: '',
   262→          newPassword: '',
   263→          confirmPassword: '',
   264→        }));
   265→      } else {
   266→        const err = await res.json().catch(() => ({}));
   267→        toast.error(err.error || 'Failed to save account');
   268→      }
   269→    } catch {
   270→      toast.error('Failed to save account');
   271→    } finally {
   272→      setAccSaving(false);
   273→    }
   274→  };
   275→
   276→  // Logo upload
   277→  const handleLogoUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   278→    const file = e.target.files?.[0];
   279→    if (!file || !currentBusiness?.id || !token) return;
   280→    setUploading(true);
   281→    try {
   282→      const formData = new FormData();
   283→      formData.append('file', file);
   284→      formData.append('businessId', currentBusiness.id);
   285→      formData.append('type', 'logo');
   286→      const res = await fetch('/api/upload', {
   287→        method: 'POST',
   288→        headers: { Authorization: `Bearer ${token}` },
   289→        body: formData,
   290→      });
   291→      if (res.ok) {
   292→        const data = await res.json();
   293→        if (data.url) {
   294→          // Persist logo to database immediately
   295→          const putRes = await fetch(`/api/businesses/${currentBusiness.id}`, {
   296→            method: 'PUT',
   297→            headers: {
   298→              Authorization: `Bearer ${token}`,
   299→              'Content-Type': 'application/json',
   300→            },
   301→            body: JSON.stringify({ logo: data.url }),
   302→          });
   303→          if (putRes.ok) {
   304→            const putData = await putRes.json();
   305→            const updatedBiz = putData.business;
   306→            setCurrentBusiness({
   307→              ...currentBusiness,
   308→              logo: updatedBiz.logo || data.url,
   309→            });
   310→            toast.success('Logo uploaded and saved');
   311→          } else {
   312→            setCurrentBusiness({
   313→              ...currentBusiness,
   314→              logo: data.url,
   315→            });
   316→            toast.success('Logo uploaded (save may be pending)');
   317→          }
   318→        }
   319→      } else {
   320→        toast.error('Failed to upload logo');
   321→      }
   322→    } catch {
   323→      toast.error('Failed to upload logo');
   324→    } finally {
   325→      setUploading(false);
   326→      if (fileInputRef.current) fileInputRef.current.value = '';
   327→    }
   328→  };
   329→
   330→  // Delete account
   331→  const handleDeleteAccount = async () => {
   332→    if (!token) return;
   333→    try {
   334→      const res = await fetch('/api/auth/me', {
   335→        method: 'DELETE',
   336→        headers: { Authorization: `Bearer ${token}` },
   337→      });
   338→      if (res.ok) {
   339→        toast.success('Account deleted');
   340→        useAuthStore.getState().logout();
   341→      } else {
   342→        toast.error('Failed to delete account');
   343→      }
   344→    } catch {
   345→      toast.error('Failed to delete account');
   346→    }
   347→  };
   348→
   349→  if (!currentBusiness) {
   350→    return (
   351→      <div className="flex flex-col items-center justify-center py-20 gap-3 text-muted-foreground">
   352→        <p>Select a business to manage settings</p>
   353→      </div>
   354→    );
   355→  }
   356→
   357→  const hasBizChanges =
   358→    bizInitial &&
   359→    (bizForm.name !== bizInitial.name ||
   360→      bizForm.category !== (bizInitial.category || '') ||
   361→      bizForm.phone !== (bizInitial.phone || '') ||
   362→      bizForm.whatsapp !== (bizInitial.whatsapp || '') ||
   363→      bizForm.address !== (bizInitial.address || '') ||
   364→      bizForm.openingHours !== (bizInitial.openingHours || '') ||
   365→      bizForm.description !== (bizInitial.description || '') ||
   366→      bizForm.status !== bizInitial.status ||
   367→      bizForm.whatsappOrder !== (bizInitial.whatsappOrder ?? false) ||
   368→      bizForm.seoEnabled !== (bizInitial.seoEnabled ?? false));
   369→
   370→return (
   371→    <>
   372→    <Tabs defaultValue={initialTab} className="space-y-6">
   373→      <TabsList>
   374→        <TabsTrigger value="business">Business Settings</TabsTrigger>
   375→        <TabsTrigger value="flipbook">Flipbook</TabsTrigger>
   376→        <TabsTrigger value="ordering">Ordering</TabsTrigger>
   377→        <TabsTrigger value="account">Account Settings</TabsTrigger>
   378→      </TabsList>
   379→
   380→      {/* Business Settings Tab */}
   381→      <TabsContent value="business" className="space-y-6">
   382→        {/* Basic Info */}
   383→        <Card>
   384→          <CardHeader>
   385→            <CardTitle>Basic Information</CardTitle>
   386→            <CardDescription>
   387→              Update your business details and branding
   388→            </CardDescription>
   389→          </CardHeader>
   390→          <CardContent className="space-y-4">
   391→            {bizLoading ? (
   392→              <div className="space-y-4">
   393→                {[1, 2, 3, 4].map((i) => (
   394→                  <div key={i} className="space-y-2">
   395→                    <div className="h-4 w-24 rounded bg-muted" />
   396→                    <div className="h-9 rounded-md bg-muted" />
   397→                  </div>
   398→                ))}
   399→              </div>
   400→            ) : (
   401→              <>
   402→                {/* Logo Upload */}
   403→                <div className="space-y-2">
   404→                  <Label>Logo</Label>
   405→                  <div className="flex items-center gap-4">
   406→                    {currentBusiness.logo ? (
   407→                      <img
   408→                        src={currentBusiness.logo}
   409→                        alt="Logo"
   410→                        className="h-16 w-16 rounded-xl object-cover ring-1 ring-border"
   411→                      />
   412→                    ) : (
   413→                      <div className="flex h-16 w-16 items-center justify-center rounded-xl bg-muted">
   414→                        <ImageIcon className="h-6 w-6 text-muted-foreground" />
   415→                      </div>
   416→                    )}
   417→                    <div className="flex flex-col gap-2">
   418→                      <Button
   419→                        type="button"
   420→                        variant="outline"
   421→                        size="sm"
   422→                        onClick={() => fileInputRef.current?.click()}
   423→                        disabled={uploading}
   424→                      >
   425→                        {uploading ? (
   426→                          <Loader2 className="mr-1.5 h-4 w-4 animate-spin" />
   427→                        ) : (
   428→                          <Upload className="mr-1.5 h-4 w-4" />
   429→                        )}
   430→                        {uploading ? 'Uploading...' : 'Upload Logo'}
   431→                      </Button>
   432→                      <input
   433→                        ref={fileInputRef}
   434→                        type="file"
   435→                        accept="image/*"
   436→                        onChange={handleLogoUpload}
   437→                        className="hidden"
   438→                      />
   439→                      <p className="text-xs text-muted-foreground">
   440→                        PNG, JPG up to 2MB
   441→                      </p>
   442→                    </div>
   443→                  </div>
   444→                </div>
   445→
   446→                <Separator />
   447→
   448→                {/* Business Name */}
   449→                <div className="space-y-2">
   450→                  <Label htmlFor="biz-name">Business Name</Label>
   451→                  <Input
   452→                    id="biz-name"
   453→                    value={bizForm.name}
   454→                    onChange={(e) =>
   455→                      setBizForm((p) => ({ ...p, name: e.target.value }))
   456→                    }
   457→                    placeholder="Enter business name"
   458→                  />
   459→                </div>
   460→
   461→                {/* Public URL / Slug */}
   462→                <div className="space-y-2">
   463→                  <Label htmlFor="biz-slug">Public URL</Label>
   464→                  <div className="flex items-center gap-0">
   465→                    <div className="flex items-center bg-muted text-muted-foreground text-sm px-3 py-2 rounded-l-md border border-r-0 border-input">
   466→                      {getAppUrl()}/p/
   467→                    </div>
   468→                    <Input
   469→                      id="biz-slug"
   470→                      value={currentBusiness?.slug || ''}
   471→                      disabled
   472→                      className="rounded-l-none"
   473→                      placeholder="auto-generated-slug"
   474→                    />
   475→                  </div>
   476→                  <p className="text-xs text-muted-foreground">
   477→                    Slug is auto-generated from your business name. Contact support to change it.
   478→                  </p>
   479→                </div>
   480→
   481→                {/* Category */}
   482→                <div className="space-y-2">
   483→                  <Label>Category</Label>
   484→                  <Select
   485→                    value={bizForm.category}
   486→                    onValueChange={(v) =>
   487→                      setBizForm((p) => ({ ...p, category: v }))
   488→                    }
   489→                  >
   490→                    <SelectTrigger className="w-full">
   491→                      <SelectValue placeholder="Select a category" />
   492→                    </SelectTrigger>
   493→                    <SelectContent>
   494→                      {BUSINESS_CATEGORIES.map((cat) => (
   495→                        <SelectItem key={cat} value={cat}>
   496→                          {cat}
   497→                        </SelectItem>
   498→                      ))}
   499→                    </SelectContent>
   500→                  </Select>
   501→                </div>
   502→
   503→                {/* Phone & WhatsApp */}
   504→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   505→                  <div className="space-y-2">
   506→                    <Label htmlFor="biz-phone">Phone</Label>
   507→                    <Input
   508→                      id="biz-phone"
   509→                      value={bizForm.phone}
   510→                      onChange={(e) =>
   511→                        setBizForm((p) => ({ ...p, phone: e.target.value }))
   512→                      }
   513→                      placeholder="Phone number"
   514→                    />
   515→                  </div>
   516→                  <div className="space-y-2">
   517→                    <Label htmlFor="biz-whatsapp">WhatsApp</Label>
   518→                    <Input
   519→                      id="biz-whatsapp"
   520→                      value={bizForm.whatsapp}
   521→                      onChange={(e) =>
   522→                        setBizForm((p) => ({ ...p, whatsapp: e.target.value }))
   523→                      }
   524→                      placeholder="WhatsApp number"
   525→                    />
   526→                  </div>
   527→                </div>
   528→
   529→                {/* Address */}
   530→                <div className="space-y-2">
   531→                  <Label htmlFor="biz-address">Address</Label>
   532→                  <Input
   533→                    id="biz-address"
   534→                    value={bizForm.address}
   535→                    onChange={(e) =>
   536→                      setBizForm((p) => ({ ...p, address: e.target.value }))
   537→                    }
   538→                    placeholder="Business address"
   539→                  />
   540→                </div>
   541→
   542→                {/* Opening Hours */}
   543→                <div className="space-y-2">
   544→                  <Label htmlFor="biz-hours">Opening Hours</Label>
   545→                  <Input
   546→                    id="biz-hours"
   547→                    value={bizForm.openingHours}
   548→                    onChange={(e) =>
   549→                      setBizForm((p) => ({ ...p, openingHours: e.target.value }))
   550→                    }
   551→                    placeholder="e.g. Mon-Fri 9AM-10PM, Sat-Sun 10AM-11PM"
   552→                  />
   553→                </div>
   554→
   555→                {/* Description */}
   556→                <div className="space-y-2">
   557→                  <Label htmlFor="biz-desc">Description / Tagline</Label>
   558→                  <Textarea
   559→                    id="biz-desc"
   560→                    value={bizForm.description}
   561→                    onChange={(e) =>
   562→                      setBizForm((p) => ({ ...p, description: e.target.value }))
   563→                    }
   564→                    placeholder="Brief description of your business"
   565→                    rows={3}
   566→                  />
   567→                </div>
   568→
   569→                {/* Welcome Message */}
   570→                <div className="space-y-2">
   571→                  <Label htmlFor="biz-welcome">Welcome Message</Label>
   572→                  <Textarea
   573→                    id="biz-welcome"
   574→                    value={(bizForm as Record<string, unknown>).welcomeMessage as string || ''}
   575→                    onChange={(e) =>
   576→                      setBizForm((p) => ({ ...p, welcomeMessage: e.target.value }))
   577→                    }
   578→                    placeholder="Welcome message shown to customers when they open your menu. e.g. Thank you for visiting us. Explore our menu and discover our products."
   579→                    rows={3}
   580→                  />
   581→                  <p className="text-xs text-muted-foreground">
   582→                    This message appears on the Welcome page of your digital book experience.
   583→                  </p>
   584→                </div>
   585→              </>
   586→            )}
   587→          </CardContent>
   588→        </Card>
   589→
   590→        {/* Menu Settings */}
   591→        <Card>
   592→          <CardHeader>
   593→            <CardTitle>Menu Settings</CardTitle>
   594→            <CardDescription>
   595→              Control how your menu is published and accessed
   596→            </CardDescription>
   597→          </CardHeader>
   598→          <CardContent className="space-y-6">
   599→            {bizLoading ? (
   600→              <div className="space-y-4">
   601→                {[1, 2, 3].map((i) => (
   602→                  <div key={i} className="flex items-center justify-between">
   603→                    <div className="space-y-1">
   604→                      <div className="h-4 w-32 rounded bg-muted" />
   605→                      <div className="h-3 w-48 rounded bg-muted" />
   606→                    </div>
   607→                    <div className="h-5 w-10 rounded-full bg-muted" />
   608→                  </div>
   609→                ))}
   610→              </div>
   611→            ) : (
   612→              <>
   613→                {/* Menu Status */}
   614→                <div className="space-y-2">
   615→                  <Label>Menu Status</Label>
   616→                  <Select
   617→                    value={bizForm.status}
   618→                    onValueChange={(v) =>
   619→                      setBizForm((p) => ({ ...p, status: v }))
   620→                    }
   621→                  >
   622→                    <SelectTrigger className="w-full sm:w-64">
