    41→/*  Template helpers                                                   */
    42→/* ------------------------------------------------------------------ */
    43→type TemplateName =
    44→  | 'modern' | 'classic' | 'luxury' | 'minimal' | 'fastfood'
    45→  | 'cafe' | 'pizza' | 'dark' | 'colorful' | 'elegant';
    46→
    47→function getTemplateClasses(template: string, primaryColor: string): string {
    48→  const t = template as TemplateName;
    49→  const base = 'min-h-screen';
    50→  const overrides: Record<TemplateName, string> = {
    51→    modern: 'bg-white text-gray-900',
    52→    classic: 'bg-amber-50/50 text-gray-900 font-serif',
    53→    luxury: 'bg-gray-950 text-gray-100',
    54→    minimal: 'bg-white text-gray-900',
    55→    fastfood: 'bg-yellow-50 text-gray-900',
    56→    cafe: 'bg-orange-50/40 text-gray-900',
    57→    pizza: 'bg-red-50/30 text-gray-900',
    58→    dark: 'bg-[#1a1a2e] text-gray-100',
    59→    colorful: 'bg-white text-gray-900',
    60→    elegant: 'bg-stone-50 text-gray-800',
    61→  };
    62→  return `${base} ${overrides[t] ?? overrides.modern}`;
    63→}
    64→
    65→function getHeaderBg(template: string, primaryColor: string): string {
    66→  const t = template as TemplateName;
    67→  const map: Record<TemplateName, string> = {
    68→    modern:   `bg-gradient-to-br from-white via-white to-gray-50`,
    69→    classic:  `bg-gradient-to-br from-amber-100 via-amber-50 to-orange-50`,
    70→    luxury:   `bg-gradient-to-br from-gray-900 via-gray-800 to-gray-900`,
    71→    minimal:  `bg-white border-b border-gray-200`,
    72→    fastfood: `bg-gradient-to-br from-yellow-400 via-orange-400 to-red-400`,
    73→    cafe:     `bg-gradient-to-br from-amber-800 via-amber-700 to-yellow-800`,
    74→    pizza:    `bg-gradient-to-br from-green-700 via-green-600 to-red-700`,
    75→    dark:     `bg-gradient-to-br from-[#1a1a2e] via-[#16213e] to-[#0f3460]`,
    76→    colorful: `bg-gradient-to-r from-purple-500 via-pink-500 to-orange-400`,
    77→    elegant:  `bg-gradient-to-br from-stone-100 via-stone-50 to-stone-100`,
    78→  };
    79→  return map[t] ?? map.modern;
    80→}
    81→
    82→function getHeaderTextColor(template: string): string {
    83→  const dark = ['luxury', 'dark', 'fastfood', 'cafe', 'pizza', 'colorful'];
    84→  return dark.includes(template) ? 'text-white' : 'text-gray-900';
    85→}
    86→
    87→function getCategoryBtnStyle(template: string, primaryColor: string, isActive: boolean): React.CSSProperties {
    88→  const t = template as TemplateName;
    89→  if (isActive) {
    90→    if (t === 'luxury' || t === 'dark') return { backgroundColor: primaryColor, color: '#fff' };
    91→    if (t === 'colorful') return { background: `linear-gradient(135deg, ${primaryColor}, #ec4899)`, color: '#fff' };
    92→    return { backgroundColor: primaryColor, color: '#fff' };
    93→  }
    94→  if (t === 'luxury' || t === 'dark') return { backgroundColor: 'rgba(255,255,255,0.08)', color: 'rgba(255,255,255,0.6)' };
    95→  return { backgroundColor: '#f3f4f6', color: '#374151' };
    96→}

   159→        return prev.map(b => b.id === item.id ? { ...b, quantity: b.quantity + 1 } : b);

   184→    const message = encodeURIComponent(`${greeting}\n\n${itemsStr}${totalStr}`);

   188→  const primaryColor = business.primaryColor || '#10b981';
   189→  const secondaryColor = business.secondaryColor || '#059669';
   190→  const templateName = business.templateName || 'modern';

   195→    playfair: '"Playfair Display", Georgia, serif',

   201→  /* ------------------------------------------------------------------ */

   203→    return (
   204→      <div className="min-h-screen bg-gray-50 flex items-center justify-center px-4">
   205→        <div className="text-center max-w-sm">
   206→          <div className="w-16 h-16 rounded-2xl bg-amber-100 flex items-center justify-center mx-auto mb-4">
   207→            <ChefHat className="w-8 h-8 text-amber-600" />

   210→          <p className="text-gray-500 text-sm">
   211→            This business is updating their digital catalog. Please check back soon.
   212→          </p>
   213→        </div>

   285→          business={business}
   286→          categories={categories}

   288→          isPreview={false}

   292→  }

   296→  /* ------------------------------------------------------------------ */
   297→  const headerTextColor = getHeaderTextColor(templateName);
   298→
   299→  // If there's a published upload, show it in a simple viewer (non-flipbook list mode)
   300→  if (hasPublishedUpload && publishedUpload && !business.flipbookEnabled) {
   301→    const isImage = publishedUpload.fileType?.startsWith('image/');
   302→    return (
   303→      <div className={getTemplateClasses(templateName, primaryColor)} style={{ fontFamily }}>
   304→        {/* Header */}
   305→        <div className={`${getHeaderBg(templateName, primaryColor)} ${headerTextColor} px-6 pt-8 pb-6`}>
   306→          <div className="max-w-2xl mx-auto text-center">
   307→            {business.logo ? (
   308→              <img src={business.logo} alt={business.name} className="w-20 h-20 rounded-2xl object-cover mx-auto mb-4 shadow-lg ring-4 ring-white/20" />
   309→            ) : (
   310→              <div className="w-20 h-20 rounded-2xl flex items-center justify-center mx-auto mb-4 text-2xl font-bold shadow-lg ring-4 ring-white/20"
   311→                style={{ backgroundColor: `${primaryColor}30`, color: headerTextColor === 'text-white' ? '#fff' : primaryColor }}>
   312→                {business.name.charAt(0).toUpperCase()}

   314→            )}
   315→            <h1 className="text-2xl font-bold mb-1">{business.name}</h1>

   317→              <p className={`text-sm opacity-80 line-clamp-2 max-w-md mx-auto ${headerTextColor === 'text-white' ? 'text-white/70' : 'text-gray-500'}`}>
   318→                {business.description}

   320→            )}
   321→          </div>
   322→        </div>
   323→
   324→        {/* Uploaded Menu Content */}
   325→        <div className="max-w-2xl mx-auto p-4">
   326→          <h2 className="text-lg font-bold mb-3 text-gray-900">Our Menu</h2>

   330→            <div className="rounded-xl border border-gray-200 p-8 text-center">
   331→              <p className="text-gray-500 mb-3">{publishedUpload.fileName}</p>
   332→              <a href={publishedUpload.url} target="_blank" rel="noopener noreferrer"
   333→                className="inline-flex items-center gap-2 px-4 py-2 rounded-lg text-white text-sm font-medium"

   336→                View PDF Menu
   337→              </a>

   340→        </div>
   341→
   342→        {/* Footer */}
   343→        <div className="text-center py-6 text-xs text-gray-400 border-t">
   344→          MADE BY <span className="font-semibold">ALTECH</span>
   345→        </div>
   346→      </div>
   347→    );
   348→  }

   350→  return (
   351→    <div className={getTemplateClasses(templateName, primaryColor)} style={{ fontFamily }}>
   352→      {/* View mode switcher */}
   353→      {business.flipbookEnabled && (
   354→        <div className="fixed top-3 right-3 z-50 flex gap-1 p-1 bg-white/90 backdrop-blur rounded-lg shadow-sm">
   355→          <button
   356→            onClick={() => setViewMode('flipbook')}
   357→            className="px-3 py-1.5 text-xs font-medium rounded-md bg-gray-100 text-gray-600 hover:bg-gray-200 transition-colors"
   358→          >
   359→            Flipbook
   360→          </button>
   361→          <button
   362→            onClick={() => setViewMode('list')}
   363→            className="px-3 py-1.5 text-xs font-medium rounded-md transition-colors"
   364→            style={{ backgroundColor: primaryColor, color: '#fff' }}
   365→          >
   366→            List
   367→          </button>
   368→        </div>
   369→      )}
   370→
   371→      {/* Header */}

   409→                <MessageCircle className="h-3 w-3" />
   410→                Order via WhatsApp
   411→              </span>
   412→            )}
   413→          </div>
