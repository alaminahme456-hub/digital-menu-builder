   200→/* ------------------------------------------------------------------ */
   201→/*  Opening hours helper                                               */
   202→/* ------------------------------------------------------------------ */
   203→function isOpenNow(openingHours: string | null): boolean {
   204→  if (!openingHours) return true;
   205→  try {
   206→    const parsed = JSON.parse(openingHours);
   207→    if (!Array.isArray(parsed) || parsed.length === 0) return true;
   208→    const now = new Date();
   209→    const dayIndex = now.getDay() === 0 ? 6 : now.getDay() - 1;
   210→    const todayEntry = parsed[dayIndex];
   211→    if (!todayEntry || !todayEntry.open || todayEntry.closed) return false;
   212→    const [openH, openM] = todayEntry.open.split(':').map(Number);
   213→    const [closeH, closeM] = todayEntry.close.split(':').map(Number);
   214→    const currentMinutes = now.getHours() * 60 + now.getMinutes();
   215→    const openMinutes = openH * 60 + openM;
   216→    const closeMinutes = closeH * 60 + closeM;
   217→    return currentMinutes >= openMinutes && currentMinutes < closeMinutes;
   218→  } catch {
   219→    return true;
   220→  }
   221→}

   303→      cats = cats.filter((c) => c.id === activeCategory);

   310→          items: (c.items ?? []).filter(
   311→            (i) =>

   315→        }))

   326→        return prev.map((o) =>
   327→          o.name === item.name ? { ...o, quantity: o.quantity + 1 } : o

   329→      }
   330→      return [...prev, { name: item.name, price: item.price, quantity: 1 }];
