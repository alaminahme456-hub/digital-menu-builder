    56→  coverImage,

    60→  onTapOpen,

    63→  businessName: string;

    69→  onTapOpen?: () => void;

    72→  const accent = accentOverride || s.accentColor;
    73→  const isCard = size === 'card';
    74→  const isPreview = size === 'preview';
    75→

    77→    serif: "'Playfair Display', Georgia, serif",

   106→      {/* Overlay */}
   107→      {s.overlay === 'dark-gradient' && (
   108→        <div className="absolute inset-0" style={{ background: 'linear-gradient(to bottom, transparent 40%, rgba(0,0,0,0.6))', pointerEvents: 'none' }} />
   109→      )}
   110→      {s.overlay === 'light-gradient' && (
   111→        <div className="absolute inset-0" style={{ background: 'linear-gradient(to bottom, transparent 40%, rgba(255,255,255,0.6))', pointerEvents: 'none' }} />
   112→      )}
   113→      {s.overlay === 'solid-bottom' && (
   114→        <div className="absolute bottom-0 left-0 right-0 h-1/3" style={{ background: isDark ? 'rgba(0,0,0,0.7)' : 'rgba(255,255,255,0.9)', pointerEvents: 'none' }} />
   115→      )}

   117→        <div className="absolute inset-0" style={{ background: 'radial-gradient(ellipse at center, transparent 50%, rgba(0,0,0,0.4))', pointerEvents: 'none' }} />
   118→      )}
   119→
   120→      {/* Pattern overlay */}
   121→      {s.pattern === 'diamonds' && (
   122→        <div className="absolute inset-0 opacity-[0.03]" style={{
   123→          backgroundImage: `repeating-linear-gradient(45deg, ${accent} 0px, ${accent} 1px, transparent 1px, transparent 20px),
   124→            repeating-linear-gradient(-45deg, ${accent} 0px, ${accent} 1px, transparent 1px, transparent 20px)`,
   125→          pointerEvents: 'none',
   126→        }} />
   127→      )}
   128→      {s.pattern === 'dots' && (
   129→        <div className="absolute inset-0 opacity-[0.04]" style={{
   130→          backgroundImage: `radial-gradient(circle, ${accent} 1px, transparent 1px)`,
   131→          backgroundSize: '16px 16px',
   132→          pointerEvents: 'none',
   133→        }} />
   134→      )}
   135→      {s.pattern === 'lines' && (
   136→        <div className="absolute inset-0 opacity-[0.03]" style={{
   137→          backgroundImage: `repeating-linear-gradient(0deg, ${accent} 0px, ${accent} 1px, transparent 1px, transparent 12px)`,
   138→          pointerEvents: 'none',
   139→        }} />
   140→      )}
   141→      {s.pattern === 'geometric' && (
   142→        <div className="absolute inset-0 opacity-[0.03]" style={{
   143→          backgroundImage: `linear-gradient(30deg, ${accent} 12%, transparent 12.5%, transparent 87%, ${accent} 87.5%, ${accent}),
   144→            linear-gradient(150deg, ${accent} 12%, transparent 12.5%, transparent 87%, ${accent} 87.5%, ${accent}),
   145→            linear-gradient(30deg, ${accent} 12%, transparent 12.5%, transparent 87%, ${accent} 87.5%, ${accent}),

   160→      )}
   161→
   162→      {/* Cover image */}
   163→      {coverImage && s.coverImagePosition !== 'none' ? (

   176→          <img
   177→            src={coverImage}
   178→            alt=""
   179→            className="w-full h-full object-cover"

   182→            <div className="absolute inset-0" style={{ background: isDark ? 'linear-gradient(to bottom, rgba(0,0,0,0.3) 0%, rgba(0,0,0,0.6) 100%)' : 'linear-gradient(to bottom, transparent 0%, rgba(255,255,255,0.8) 100%)' }} />

   192→              s.coverImageShape === 'arch' ? '999px 999px 0 0' : '0',

   195→        >
   196→          <ImageIcon className={`${isCard ? 'w-6 h-6' : 'w-10 h-10'} ${isDark ? 'text-white/10' : 'text-black/10'}`} />

   200→      {/* Content area */}

   213→            style={{ color: accent, fontFamily: subFont }}

   216→              <span className="flex items-center gap-1">
   217→                <span className="inline-block w-3 h-px" style={{ background: accent }} />
   218→                {s.tagName}
   219→                <span className="inline-block w-3 h-px" style={{ background: accent }} />

   299→          >
   300→            {s.buttonLabel}
   301→          </button>
   302→        )}
   303→      </div>
   304→    </div>
   305→  );
   306→}

   414→// COVER CUSTOMIZATION PANEL
   415→// ═══════════════════════════════════════════════════════

   417→function CoverCustomizationPanel({
   418→  coverImage,
   419→  coverTagline,
   420→  coverAccent,
   421→  onChangeImage,
   422→  onChangeTagline,
   423→  onChangeAccent,
   424→  onRemoveImage,
   425→}: {
   426→  coverImage: string | null;
   427→  coverTagline: string;
   428→  coverAccent: string;
   429→  onChangeImage: (url: string | null) => void;
   430→  onChangeTagline: (val: string) => void;
   431→  onChangeAccent: (val: string) => void;
   432→  onRemoveImage: () => void;
   433→}) {
   434→  const [uploading, setUploading] = useState(false);
   435→  const fileInputRef = useRef<HTMLInputElement>(null);
   436→  const token = useAuthStore((s) => s.token);
   437→
   438→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   439→    const file = e.target.files?.[0];
   440→    if (!file || !token) return;
   441→    setUploading(true);
   442→    try {
   443→      const formData = new FormData();
   444→      formData.append('file', file);
   445→      const res = await fetch('/api/upload', {
   446→        method: 'POST',
   447→        headers: { Authorization: `Bearer ${token}` },
   448→        body: formData,
   449→      });
   450→      if (res.ok) {
   451→        const data = await res.json();
   452→        onChangeImage(data.url);
   453→        toast.success('Cover image uploaded');
   454→      } else {
   455→        toast.error('Failed to upload image');
   456→      }
   457→    } catch {
   458→      toast.error('Upload failed');
   459→    } finally {
   460→      setUploading(false);
   461→    }
   462→  };
   463→
   464→  const accentPresets = [
   465→    { label: 'Gold', value: '#C9A84C' },
   466→    { label: 'Silver', value: '#A1A1AA' },
   467→    { label: 'Black', value: '#0A0A0A' },
   468→    { label: 'White', value: '#FFFFFF' },
   469→    { label: 'Rose', value: '#E11D48' },
   470→    { label: 'Ocean', value: '#0891B2' },
   471→    { label: 'Emerald', value: '#059669' },
   472→    { label: 'Violet', value: '#7C3AED' },
   473→  ];
   474→
   475→  return (
   476→    <div className="space-y-5">
   477→      {/* Cover Image */}
   478→      <div>
   479→        <label className="flex items-center gap-2 text-[12px] font-semibold text-charcoal/70 uppercase tracking-wider mb-2">
   480→          <ImageIcon className="w-3.5 h-3.5" />
   481→          Cover Image
   482→        </label>
   483→        {coverImage ? (
   484→          <div className="relative rounded-xl overflow-hidden aspect-[3/2] bg-smoke">
   485→            <img src={coverImage} alt="Cover" className="w-full h-full object-cover" />
   486→            <div className="absolute inset-0 bg-black/0 hover:bg-black/40 transition-colors flex items-center justify-center gap-2 opacity-0 hover:opacity-100">
