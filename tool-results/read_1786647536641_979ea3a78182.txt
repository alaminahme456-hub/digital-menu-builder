     1→'use client';
     2→
     3→import { useState, useEffect, useRef } from 'react';
     4→import {
     5→  Save,
     6→  Upload,
     7→  Trash2,
     8→  Loader2,
     9→  ImageIcon,
    10→  AlertTriangle,
    11→} from 'lucide-react';
    12→import {
    13→  Card,
    14→  CardContent,
    15→  CardHeader,
    16→  CardTitle,
    17→  CardDescription,
    18→} from '@/components/ui/card';
    19→import { Button } from '@/components/ui/button';
    20→import { Input } from '@/components/ui/input';
    21→import { Textarea } from '@/components/ui/textarea';
    22→import { Label } from '@/components/ui/label';
    23→import { Switch } from '@/components/ui/switch';
    24→import { Separator } from '@/components/ui/separator';
    25→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    26→import {
    27→  Select,
    28→  SelectContent,
    29→  SelectItem,
    30→  SelectTrigger,
    31→  SelectValue,
    32→} from '@/components/ui/select';
    33→import {
    34→  AlertDialog,
    35→  AlertDialogAction,
    36→  AlertDialogCancel,
    37→  AlertDialogContent,
    38→  AlertDialogDescription,
    39→  AlertDialogFooter,
    40→  AlertDialogHeader,
    41→  AlertDialogTitle,
    42→  AlertDialogTrigger,
    43→} from '@/components/ui/alert-dialog';
    44→import { useAuthStore, useAppStore } from '@/lib/store';
    45→import type { Business } from '@/lib/types';
    46→import { BUSINESS_CATEGORIES } from '@/lib/types';
    47→import { toast } from 'sonner';
    48→
    49→interface SettingsPanelProps {
    50→  initialTab?: string;
    51→}
    52→
    53→export default function SettingsPanel({ initialTab = 'business' }: SettingsPanelProps) {
    54→  const { token, user } = useAuthStore();
    55→  const { currentBusiness, setCurrentBusiness } = useAppStore();
    56→  const fileInputRef = useRef<HTMLInputElement>(null);
    57→
    58→  // Business form state
    59→  const [bizForm, setBizForm] = useState({
    60→    name: '',
    61→    category: '',
    62→    phone: '',
    63→    whatsapp: '',
    64→    address: '',
    65→    openingHours: '',
    66→    description: '',
    67→    status: 'draft' as string,
    68→    whatsappOrder: false,
    69→    seoEnabled: false,
    70→  });
    71→  const [bizLoading, setBizLoading] = useState(false);
    72→  const [bizSaving, setBizSaving] = useState(false);
    73→  const [bizInitial, setBizInitial] = useState<Business | null>(null);
    74→
    75→  // Account form state
    76→  const [accForm, setAccForm] = useState({
    77→    name: '',
    78→    phone: '',
    79→    currentPassword: '',
    80→    newPassword: '',
    81→    confirmPassword: '',
    82→  });
    83→  const [accSaving, setAccSaving] = useState(false);
    84→  const [accLoading, setAccLoading] = useState(true);
    85→
    86→  // Upload state
    87→  const [uploading, setUploading] = useState(false);
    88→
    89→  // Flipbook settings state
    90→  const [flipbookForm, setFlipbookForm] = useState({
    91→    flipbookEnabled: true,
    92→    flipbookAnimEnabled: true,
    93→    flipbookAnimSpeed: 'medium' as string,
    94→    flipbookPageNumbers: true,
    95→    flipbookSwipeNav: true,
    96→    flipbookSoundEffects: false,
    97→    flipbookFullscreen: true,
    98→    flipbookInteractions: true,
    99→  });
   100→  const [flipbookSaving, setFlipbookSaving] = useState(false);
   101→
   102→  // Ordering settings state
   103→  const [orderingForm, setOrderingForm] = useState({
   104→    whatsappOrder: true,
   105→    basketEnabled: true,
   106→    showQuantitySelector: true,
   107→    showOrderButton: true,
   108→    whatsappGreeting: 'Hello, I would like to place an order:',
   109→  });
   110→  const [orderingSaving, setOrderingSaving] = useState(false);
   111→
   112→  // Fetch business data
   113→  useEffect(() => {
   114→    if (!currentBusiness?.id || !token) return;
   115→    setBizLoading(true);
   116→    async function fetchBusiness() {
   117→      try {
   118→        const res = await fetch(`/api/businesses/${currentBusiness.id}`, {
   119→          headers: { Authorization: `Bearer ${token}` },
   120→        });
   121→        if (res.ok) {
   122→          const data = await res.json();
   123→          const b: Business = data.business;
   124→          setBizInitial(b);
   125→          setBizForm({
   126→            name: b.name || '',
   127→            category: b.category || '',
   128→            phone: b.phone || '',
   129→            whatsapp: b.whatsapp || '',
   130→            address: b.address || '',
   131→            openingHours: b.openingHours || '',
   132→            description: b.description || '',
   133→            status: b.status || 'draft',
   134→            whatsappOrder: b.whatsappOrder ?? false,
   135→            seoEnabled: b.seoEnabled ?? false,
   136→          });
   137→          // Load flipbook settings
   138→          setFlipbookForm({
   139→            flipbookEnabled: b.flipbookEnabled ?? true,
   140→            flipbookAnimEnabled: b.flipbookAnimEnabled ?? true,
   141→            flipbookAnimSpeed: b.flipbookAnimSpeed || 'medium',
   142→            flipbookPageNumbers: b.flipbookPageNumbers ?? true,
   143→            flipbookSwipeNav: b.flipbookSwipeNav ?? true,
   144→            flipbookSoundEffects: b.flipbookSoundEffects ?? false,
   145→            flipbookFullscreen: b.flipbookFullscreen ?? true,
   146→            flipbookInteractions: b.flipbookInteractions ?? true,
   147→          });
   148→          // Load ordering settings
   149→          setOrderingForm({
   150→            whatsappOrder: b.whatsappOrder ?? true,
   151→            basketEnabled: b.basketEnabled ?? true,
   152→            showQuantitySelector: b.showQuantitySelector ?? true,
   153→            showOrderButton: b.showOrderButton ?? true,
   154→            whatsappGreeting: b.whatsappGreeting || 'Hello, I would like to place an order:',
   155→          });
   156→        }
   157→      } catch {
   158→        toast.error('Failed to load business settings');
   159→      } finally {
   160→        setBizLoading(false);
   161→      }
   162→    }
   163→    fetchBusiness();
   164→  }, [currentBusiness?.id, token]);
   165→
   166→  // Fetch user data
   167→  useEffect(() => {
   168→    if (!user || !token) return;
   169→    setAccForm({
   170→      name: user.name || '',
   171→      phone: '',
   172→      currentPassword: '',
   173→      newPassword: '',
   174→      confirmPassword: '',
   175→    });
   176→    setAccLoading(false);
   177→  }, [user, token]);
   178→
   179→  // Save business settings
   180→  const handleSaveBusiness = async () => {
   181→    if (!currentBusiness?.id || !token) return;
   182→    setBizSaving(true);
   183→    try {
   184→      const res = await fetch(`/api/businesses/${currentBusiness.id}`, {
   185→        method: 'PUT',
   186→        headers: {
   187→          Authorization: `Bearer ${token}`,
   188→          'Content-Type': 'application/json',
   189→        },
   190→        body: JSON.stringify(bizForm),
   191→      });
   192→      if (res.ok) {
   193→        const data = await res.json();
   194→        const updated: Business = data.business;
   195→        setBizInitial(updated);
   196→        setCurrentBusiness({
   197→          id: updated.id,
   198→          slug: updated.slug,
   199→          name: updated.name,
   200→          logo: updated.logo,
   201→          status: updated.status,
   202→        });
   203→        toast.success('Business settings saved');
   204→      } else {
   205→        const err = await res.json().catch(() => ({}));
   206→        toast.error(err.error || 'Failed to save settings');
   207→      }
   208→    } catch {
   209→      toast.error('Failed to save settings');
   210→    } finally {
   211→      setBizSaving(false);
   212→    }
   213→  };
   214→
   215→  // Save account settings
   216→  const handleSaveAccount = async () => {
   217→    if (!token) return;
   218→    const updateData: Record<string, string> = {};
   219→    if (accForm.name) updateData.name = accForm.name;
   220→    if (accForm.phone) updateData.phone = accForm.phone;
   221→
   222→    // Password change validation
   223→    if (accForm.newPassword) {
   224→      if (!accForm.currentPassword) {
   225→        toast.error('Current password is required to change password');
   226→        return;
   227→      }
   228→      if (accForm.newPassword !== accForm.confirmPassword) {
   229→        toast.error('New passwords do not match');
   230→        return;
   231→      }
   232→      if (accForm.newPassword.length < 8) {
   233→        toast.error('Password must be at least 8 characters');
   234→        return;
   235→      }
   236→      updateData.currentPassword = accForm.currentPassword;
   237→      updateData.newPassword = accForm.newPassword;
   238→    }
   239→
   240→    if (Object.keys(updateData).length === 0) {
   241→      toast.error('No changes to save');
   242→      return;
   243→    }
   244→
   245→    setAccSaving(true);
   246→    try {
   247→      const res = await fetch('/api/auth/me', {
   248→        method: 'PUT',
   249→        headers: {
   250→          Authorization: `Bearer ${token}`,
   251→          'Content-Type': 'application/json',
   252→        },
   253→        body: JSON.stringify(updateData),
   254→      });
   255→      if (res.ok) {
   256→        toast.success('Account settings saved');
   257→        setAccForm((prev) => ({
   258→          ...prev,
   259→          currentPassword: '',
   260→          newPassword: '',
   261→          confirmPassword: '',
   262→        }));
   263→      } else {
   264→        const err = await res.json().catch(() => ({}));
   265→        toast.error(err.error || 'Failed to save account');
   266→      }
   267→    } catch {
   268→      toast.error('Failed to save account');
   269→    } finally {
   270→      setAccSaving(false);
   271→    }
   272→  };
   273→
   274→  // Logo upload
   275→  const handleLogoUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   276→    const file = e.target.files?.[0];
   277→    if (!file || !currentBusiness?.id || !token) return;
   278→    setUploading(true);
   279→    try {
   280→      const formData = new FormData();
   281→      formData.append('file', file);
   282→      formData.append('businessId', currentBusiness.id);
   283→      formData.append('type', 'logo');
   284→      const res = await fetch('/api/upload', {
   285→        method: 'POST',
   286→        headers: { Authorization: `Bearer ${token}` },
   287→        body: formData,
   288→      });
   289→      if (res.ok) {
   290→        const data = await res.json();
   291→        if (data.url) {
   292→          setCurrentBusiness({
   293→            ...currentBusiness,
   294→            logo: data.url,
   295→          });
   296→          toast.success('Logo uploaded');
   297→        }
   298→      } else {
   299→        toast.error('Failed to upload logo');
   300→      }
   301→    } catch {
   302→      toast.error('Failed to upload logo');
   303→    } finally {
   304→      setUploading(false);
   305→      if (fileInputRef.current) fileInputRef.current.value = '';
   306→    }
   307→  };
   308→
   309→  // Delete account
   310→  const handleDeleteAccount = async () => {
   311→    if (!token) return;
   312→    try {
   313→      const res = await fetch('/api/auth/me', {
   314→        method: 'DELETE',
   315→        headers: { Authorization: `Bearer ${token}` },
   316→      });
   317→      if (res.ok) {
   318→        toast.success('Account deleted');
   319→        useAuthStore.getState().logout();
   320→      } else {
   321→        toast.error('Failed to delete account');
   322→      }
   323→    } catch {
   324→      toast.error('Failed to delete account');
   325→    }
   326→  };
   327→
   328→  if (!currentBusiness) {
   329→    return (
   330→      <div className="flex flex-col items-center justify-center py-20 gap-3 text-muted-foreground">
   331→        <p>Select a business to manage settings</p>
   332→      </div>
   333→    );
   334→  }
   335→
   336→  const hasBizChanges =
   337→    bizInitial &&
   338→    (bizForm.name !== bizInitial.name ||
   339→      bizForm.category !== (bizInitial.category || '') ||
   340→      bizForm.phone !== (bizInitial.phone || '') ||
   341→      bizForm.whatsapp !== (bizInitial.whatsapp || '') ||
   342→      bizForm.address !== (bizInitial.address || '') ||
   343→      bizForm.openingHours !== (bizInitial.openingHours || '') ||
   344→      bizForm.description !== (bizInitial.description || '') ||
   345→      bizForm.status !== bizInitial.status ||
   346→      bizForm.whatsappOrder !== (bizInitial.whatsappOrder ?? false) ||
   347→      bizForm.seoEnabled !== (bizInitial.seoEnabled ?? false));
   348→
   349→  return (
   350→    <Tabs defaultValue={initialTab} className="space-y-6">
   351→      <TabsList>
   352→        <TabsTrigger value="business">Business Settings</TabsTrigger>
   353→        <TabsTrigger value="flipbook">Flipbook</TabsTrigger>
   354→        <TabsTrigger value="ordering">Ordering</TabsTrigger>
   355→        <TabsTrigger value="account">Account Settings</TabsTrigger>
   356→      </TabsList>
   357→
   358→      {/* Business Settings Tab */}
   359→      <TabsContent value="business" className="space-y-6">
   360→        {/* Basic Info */}
   361→        <Card>
   362→          <CardHeader>
   363→            <CardTitle>Basic Information</CardTitle>
   364→            <CardDescription>
   365→              Update your business details and branding
   366→            </CardDescription>
   367→          </CardHeader>
   368→          <CardContent className="space-y-4">
   369→            {bizLoading ? (
   370→              <div className="space-y-4">
   371→                {[1, 2, 3, 4].map((i) => (
   372→                  <div key={i} className="space-y-2">
   373→                    <div className="h-4 w-24 rounded bg-muted" />
   374→                    <div className="h-9 rounded-md bg-muted" />
   375→                  </div>
   376→                ))}
   377→              </div>
   378→            ) : (
   379→              <>
   380→                {/* Logo Upload */}
   381→                <div className="space-y-2">
   382→                  <Label>Logo</Label>
   383→                  <div className="flex items-center gap-4">
   384→                    {currentBusiness.logo ? (
   385→                      <img
   386→                        src={currentBusiness.logo}
   387→                        alt="Logo"
   388→                        className="h-16 w-16 rounded-xl object-cover ring-1 ring-border"
   389→                      />
   390→                    ) : (
   391→                      <div className="flex h-16 w-16 items-center justify-center rounded-xl bg-muted">
   392→                        <ImageIcon className="h-6 w-6 text-muted-foreground" />
   393→                      </div>
   394→                    )}
   395→                    <div className="flex flex-col gap-2">
   396→                      <Button
   397→                        type="button"
   398→                        variant="outline"
   399→                        size="sm"
   400→                        onClick={() => fileInputRef.current?.click()}
   401→                        disabled={uploading}
   402→                      >
   403→                        {uploading ? (
   404→                          <Loader2 className="mr-1.5 h-4 w-4 animate-spin" />
   405→                        ) : (
   406→                          <Upload className="mr-1.5 h-4 w-4" />
   407→                        )}
   408→                        {uploading ? 'Uploading...' : 'Upload Logo'}
   409→                      </Button>
   410→                      <input
   411→                        ref={fileInputRef}
   412→                        type="file"
   413→                        accept="image/*"
   414→                        onChange={handleLogoUpload}
   415→                        className="hidden"
   416→                      />
   417→                      <p className="text-xs text-muted-foreground">
   418→                        PNG, JPG up to 2MB
   419→                      </p>
   420→                    </div>
   421→                  </div>
   422→                </div>
   423→
   424→                <Separator />
   425→
   426→                {/* Business Name */}
   427→                <div className="space-y-2">
   428→                  <Label htmlFor="biz-name">Business Name</Label>
   429→                  <Input
   430→                    id="biz-name"
   431→                    value={bizForm.name}
   432→                    onChange={(e) =>
   433→                      setBizForm((p) => ({ ...p, name: e.target.value }))
   434→                    }
   435→                    placeholder="Enter business name"
   436→                  />
   437→                </div>
   438→
   439→                {/* Category */}
   440→                <div className="space-y-2">
   441→                  <Label>Category</Label>
   442→                  <Select
   443→                    value={bizForm.category}
   444→                    onValueChange={(v) =>
   445→                      setBizForm((p) => ({ ...p, category: v }))
   446→                    }
   447→                  >
   448→                    <SelectTrigger className="w-full">
   449→                      <SelectValue placeholder="Select a category" />
   450→                    </SelectTrigger>
   451→                    <SelectContent>
   452→                      {BUSINESS_CATEGORIES.map((cat) => (
   453→                        <SelectItem key={cat} value={cat}>
   454→                          {cat}
   455→                        </SelectItem>
   456→                      ))}
   457→                    </SelectContent>
   458→                  </Select>
   459→                </div>
   460→
   461→                {/* Phone & WhatsApp */}
   462→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   463→                  <div className="space-y-2">
   464→                    <Label htmlFor="biz-phone">Phone</Label>
   465→                    <Input
   466→                      id="biz-phone"
   467→                      value={bizForm.phone}
   468→                      onChange={(e) =>
   469→                        setBizForm((p) => ({ ...p, phone: e.target.value }))
   470→                      }
   471→                      placeholder="Phone number"
   472→                    />
   473→                  </div>
   474→                  <div className="space-y-2">
   475→                    <Label htmlFor="biz-whatsapp">WhatsApp</Label>
   476→                    <Input
   477→                      id="biz-whatsapp"
   478→                      value={bizForm.whatsapp}
   479→                      onChange={(e) =>
   480→                        setBizForm((p) => ({ ...p, whatsapp: e.target.value }))
   481→                      }
   482→                      placeholder="WhatsApp number"
   483→                    />
   484→                  </div>
   485→                </div>
   486→
   487→                {/* Address */}
   488→                <div className="space-y-2">
   489→                  <Label htmlFor="biz-address">Address</Label>
   490→                  <Input
   491→                    id="biz-address"
   492→                    value={bizForm.address}
   493→                    onChange={(e) =>
   494→                      setBizForm((p) => ({ ...p, address: e.target.value }))
   495→                    }
   496→                    placeholder="Business address"
   497→                  />
   498→                </div>
   499→
   500→                {/* Opening Hours */}
   501→                <div className="space-y-2">
   502→                  <Label htmlFor="biz-hours">Opening Hours</Label>
   503→                  <Input
   504→                    id="biz-hours"
   505→                    value={bizForm.openingHours}
   506→                    onChange={(e) =>
   507→                      setBizForm((p) => ({ ...p, openingHours: e.target.value }))
   508→                    }
   509→                    placeholder="e.g. Mon-Fri 9AM-10PM, Sat-Sun 10AM-11PM"
   510→                  />
   511→                </div>
   512→
   513→                {/* Description */}
   514→                <div className="space-y-2">
   515→                  <Label htmlFor="biz-desc">Description</Label>
   516→                  <Textarea
   517→                    id="biz-desc"
   518→                    value={bizForm.description}
   519→                    onChange={(e) =>
   520→                      setBizForm((p) => ({ ...p, description: e.target.value }))
   521→                    }
   522→                    placeholder="Brief description of your business"
   523→                    rows={3}
   524→                  />
   525→                </div>
   526→              </>
   527→            )}
   528→          </CardContent>
   529→        </Card>
   530→
   531→        {/* Menu Settings */}
   532→        <Card>
   533→          <CardHeader>
   534→            <CardTitle>Menu Settings</CardTitle>
   535→            <CardDescription>
   536→              Control how your menu is published and accessed
   537→            </CardDescription>
   538→          </CardHeader>
   539→          <CardContent className="space-y-6">
   540→            {bizLoading ? (
   541→              <div className="space-y-4">
   542→                {[1, 2, 3].map((i) => (
   543→                  <div key={i} className="flex items-center justify-between">
   544→                    <div className="space-y-1">
   545→                      <div className="h-4 w-32 rounded bg-muted" />
   546→                      <div className="h-3 w-48 rounded bg-muted" />
   547→                    </div>
   548→                    <div className="h-5 w-10 rounded-full bg-muted" />
   549→                  </div>
   550→                ))}
   551→              </div>
   552→            ) : (
   553→              <>
   554→                {/* Menu Status */}
   555→                <div className="space-y-2">
   556→                  <Label>Menu Status</Label>
   557→                  <Select
   558→                    value={bizForm.status}
   559→                    onValueChange={(v) =>
   560→                      setBizForm((p) => ({ ...p, status: v }))
   561→                    }
   562→                  >
   563→                    <SelectTrigger className="w-full sm:w-64">
   564→                      <SelectValue />
   565→                    </SelectTrigger>
   566→                    <SelectContent>
   567→                      <SelectItem value="draft">Draft</SelectItem>
   568→                      <SelectItem value="published">Published</SelectItem>
   569→                      <SelectItem value="unpublished">Unpublished</SelectItem>
   570→                    </SelectContent>
   571→                  </Select>
   572→                  <p className="text-xs text-muted-foreground">
   573→                    Only published menus are accessible via QR code
   574→                  </p>
   575→                </div>
   576→
   577→                <Separator />
   578→
   579→                {/* WhatsApp Ordering */}
   580→                <div className="flex items-center justify-between gap-4">
   581→                  <div className="space-y-0.5">
   582→                    <Label>WhatsApp Ordering</Label>
   583→                    <p className="text-sm text-muted-foreground">
   584→                      Allow customers to place orders via WhatsApp
   585→                    </p>
   586→                  </div>
   587→                  <Switch
   588→                    checked={bizForm.whatsappOrder}
   589→                    onCheckedChange={(v) =>
   590→                      setBizForm((p) => ({ ...p, whatsappOrder: v }))
   591→                    }
   592→                  />
   593→                </div>
   594→
   595→                <Separator />
   596→
   597→                {/* SEO */}
   598→                <div className="flex items-center justify-between gap-4">
   599→                  <div className="space-y-0.5">
   600→                    <Label>SEO Indexing</Label>
   601→                    <p className="text-sm text-muted-foreground">
   602→                      Allow search engines to index your menu page
   603→                    </p>
   604→                  </div>
   605→                  <Switch
   606→                    checked={bizForm.seoEnabled}
   607→                    onCheckedChange={(v) =>
   608→                      setBizForm((p) => ({ ...p, seoEnabled: v }))
   609→                    }
   610→                  />
   611→                </div>
   612→              </>
   613→            )}
   614→          </CardContent>
   615→        </Card>
   616→
   617→        {/* Save Button */}
   618→        <div className="flex justify-end">
   619→          <Button
   620→            onClick={handleSaveBusiness}
   621→            disabled={bizSaving || bizLoading || !hasBizChanges}
   622→            className="bg-emerald-600 hover:bg-emerald-700"
   623→          >
   624→            {bizSaving ? (
   625→              <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   626→            ) : (
   627→              <Save className="mr-2 h-4 w-4" />
   628→            )}
   629→            Save Changes
   630→          </Button>
   631→        </div>
   632→      </TabsContent>
