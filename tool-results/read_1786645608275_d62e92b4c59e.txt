   193→function AddCategoryDialog({
   194→  open,
   195→  onOpenChange,
   196→  businessId,
   197→  token,
   198→  existingNames,
   199→  onSuccess,
   200→}: {
   201→  open: boolean;
   202→  onOpenChange: (open: boolean) => void;
   203→  businessId: string;
   204→  token: string;
   205→  existingNames: string[];
   206→  onSuccess?: () => void;
   207→}) {
   208→  const [name, setName] = useState('');
   209→  const [loading, setLoading] = useState(false);
   210→
   211→  const availablePresets = CATEGORIES_PRESETS.filter(
   212→    (p) => !existingNames.some((e) => e.toLowerCase() === p.toLowerCase())
   213→  );
   214→
   215→  const handleSubmit = async () => {
   216→    if (!name.trim()) return;
   217→    setLoading(true);
   218→    try {
   219→      const res = await fetch('/api/menu/categories', {
   220→        method: 'POST',
   221→        headers: authHeaders(token),
   222→        body: JSON.stringify({ businessId, name: name.trim() }),
   223→      });
   224→      if (!res.ok) {
   225→        const data = await res.json();
   226→        throw new Error(data.error || 'Failed to create category');
   227→      }
   228→      toast.success(`Category "${name.trim()}" created`);
   229→      setName('');
   230→      onOpenChange(false);
   231→      onSuccess?.();
   232→    } catch (err) {
   233→      toast.error(err instanceof Error ? err.message : 'Failed to create category');
   234→    } finally {
   235→      setLoading(false);
   236→    }
   237→  };
   238→
   239→  return (
   240→    <Dialog open={open} onOpenChange={(v) => { if (!v) setName(''); onOpenChange(v); }}>
   241→      <DialogContent className="sm:max-w-md">
   242→        <DialogHeader>
   243→          <DialogTitle>Add Category</DialogTitle>
   244→          <DialogDescription>
   245→            Type a name or pick from suggestions below
   246→          </DialogDescription>
   247→        </DialogHeader>
   248→        <div className="space-y-4">
   249→          <div className="space-y-2">
   250→            <Label htmlFor="cat-name">Category Name</Label>
   251→            <Input
   252→              id="cat-name"
   253→              placeholder="e.g. Main Meals"
   254→              value={name}
   255→              onChange={(e) => setName(e.target.value)}
   256→              onKeyDown={(e) => {
   257→                if (e.key === 'Enter') handleSubmit();
   258→              }}
   259→              autoFocus
   260→            />
   261→          </div>
   262→          {availablePresets.length > 0 && (
   263→            <div className="space-y-2">
   264→              <Label className="text-xs text-muted-foreground">Suggestions</Label>
   265→              <div className="flex flex-wrap gap-2">
   266→                {availablePresets.map((preset) => (
   267→                  <button
   268→                    key={preset}
   269→                    type="button"
   270→                    onClick={() => setName(preset)}
   271→                    className="inline-flex items-center rounded-full border bg-background px-3 py-1.5 text-xs font-medium text-foreground shadow-xs transition-colors hover:bg-accent hover:text-accent-foreground"
   272→                  >
   273→                    {preset}
   274→                  </button>
   275→                ))}
   276→              </div>
   277→            </div>
   278→          )}
   279→        </div>
   280→        <DialogFooter>
   281→          <Button variant="outline" onClick={() => { setName(''); onOpenChange(false); }}>
   282→            Cancel
   283→          </Button>
   284→          <Button
   285→            onClick={handleSubmit}
   286→            disabled={!name.trim() || loading}
   287→            className="bg-emerald-600 hover:bg-emerald-700"
   288→          >
   289→            {loading && <Loader2 className="size-4 animate-spin" />}
   290→            Create Category
   291→          </Button>
   292→        </DialogFooter>
   293→      </DialogContent>
   294→    </Dialog>
   295→  );
   296→}

   298→// ============================================================

   301→
   302→function ItemFormDialog({
   303→  open,
   304→  onOpenChange,
   305→  item,
   306→  categories,
   307→  businessId,
   308→  token,
   309→}: {
   310→  open: boolean;
   311→  onOpenChange: (open: boolean) => void;
   312→  item: MenuItem | null;
   313→  categories: MenuCategory[];
   314→  businessId: string;
   315→  token: string;
   316→}) {
   317→  const isEditing = !!item;
   318→  const [form, setForm] = useState<ItemFormData>(EMPTY_ITEM_FORM);
   319→  const [loading, setLoading] = useState(false);
   320→  const [uploading, setUploading] = useState(false);
   321→  const fileInputRef = useRef<HTMLInputElement>(null);
   322→  const dragCounter = useRef(0);
   323→  const [dragOver, setDragOver] = useState(false);
   324→
   325→  useEffect(() => {
   326→    if (open) {
   327→      if (item) {
   328→        setForm({
   329→          name: item.name,
   330→          description: item.description || '',
   331→          price: String(item.price),
   332→          categoryId: item.categoryId,
   333→          image: item.image,
   334→          available: item.available,
   335→        });
   336→      } else {
   337→        setForm({
   338→          ...EMPTY_ITEM_FORM,
   339→          categoryId: categories.length > 0 ? categories[0].id : '',
   340→        });
   341→      }
   342→    }
   343→  }, [open, item, categories]);
   344→
   345→  const uploadFile = async (file: File) => {
   346→    setUploading(true);
   347→    try {
   348→      const fd = new FormData();
   349→      fd.append('file', file);
   350→      fd.append('type', 'item-image');
   351→      const res = await fetch('/api/upload', {
   352→        method: 'POST',
   353→        headers: { Authorization: `Bearer ${token}` },
   354→        body: fd,
   355→      });
   356→      if (!res.ok) throw new Error('Upload failed');
   357→      const data = await res.json();
   358→      setForm((f) => ({ ...f, image: data.url }));
   359→      toast.success('Image uploaded');
   360→    } catch {
   361→      toast.error('Failed to upload image');
   362→    } finally {
   363→      setUploading(false);
   364→    }
   365→  };
   366→
   367→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   368→    const file = e.target.files?.[0];
   369→    if (file) uploadFile(file);
   370→    if (fileInputRef.current) fileInputRef.current.value = '';
   371→  };
   372→
   373→  const handleDrop = (e: React.DragEvent) => {
   374→    e.preventDefault();
   375→    dragCounter.current = 0;
   376→    setDragOver(false);
   377→    const file = e.dataTransfer.files?.[0];
   378→    if (file && file.type.startsWith('image/')) uploadFile(file);
   379→  };
   380→
   381→  const handleDragEnter = (e: React.DragEvent) => {
   382→    e.preventDefault();
   383→    dragCounter.current++;
   384→    setDragOver(true);
   385→  };
   386→
   387→  const handleDragLeave = (e: React.DragEvent) => {
   388→    e.preventDefault();
   389→    dragCounter.current--;
   390→    if (dragCounter.current <= 0) {
   391→      dragCounter.current = 0;
   392→      setDragOver(false);
   393→    }
   394→  };
   395→
   396→  const handleSubmit = async () => {
   397→    if (!form.name.trim() || !form.price || !form.categoryId) return;
   398→    setLoading(true);
   399→    try {
   400→      const body = {
   401→        businessId,
   402→        categoryId: form.categoryId,
   403→        name: form.name.trim(),
   404→        description: form.description.trim() || null,
   405→        price: Number(form.price),
   406→        image: form.image,
   407→        available: form.available,
   408→      };
   409→
   410→      let res: Response;
   411→      if (isEditing) {
   412→        res = await fetch('/api/menu/items', {
   413→          method: 'PUT',
   414→          headers: authHeaders(token),
   415→          body: JSON.stringify({ id: item!.id, ...body }),
   416→        });
   417→      } else {
   418→        res = await fetch('/api/menu/items', {
   419→          method: 'POST',
   420→          headers: authHeaders(token),
   421→          body: JSON.stringify(body),
   422→        });
   423→      }
   424→
   425→      if (!res.ok) {
   426→        const data = await res.json();
   427→        throw new Error(data.error || `Failed to ${isEditing ? 'update' : 'create'} item`);
   428→      }
   429→
   430→      toast.success(`Item ${isEditing ? 'updated' : 'created'} successfully`);
   431→      onOpenChange(false);
   432→    } catch (err) {
   433→      toast.error(err instanceof Error ? err.message : 'Operation failed');
   434→    } finally {
   435→      setLoading(false);
   436→    }
   437→  };
   438→
   439→  return (
   440→    <Dialog open={open} onOpenChange={onOpenChange}>
   441→      <DialogContent className="flex max-h-[90vh] flex-col gap-0 overflow-hidden p-0 sm:max-w-lg">
   442→        <DialogHeader className="shrink-0 px-6 pb-4 pt-6">
   443→          <DialogTitle>{isEditing ? 'Edit Item' : 'Add New Item'}</DialogTitle>
   444→          <DialogDescription>
   445→            {isEditing
   446→              ? 'Update the menu item details below'
   447→              : 'Fill in the details to add a new menu item'}
   448→          </DialogDescription>
   449→        </DialogHeader>
   450→
   451→        <Separator />
   452→
   453→        <ScrollArea className="flex-1">
   454→          <div className="space-y-5 px-6 py-5">
   455→            {/* Image Upload */}
   456→            <div className="space-y-2">
   457→              <Label>Food Image</Label>
   458→              <div
   459→                onDrop={handleDrop}
   460→                onDragEnter={handleDragEnter}
   461→                onDragLeave={handleDragLeave}
   462→                onDragOver={(e) => e.preventDefault()}
   463→                onClick={() => fileInputRef.current?.click()}
   464→                className={`relative flex cursor-pointer flex-col items-center justify-center gap-2 rounded-xl border-2 border-dashed p-6 transition-colors ${
   465→                  dragOver
   466→                    ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-950/30'
   467→                    : form.image
   468→                      ? 'border-transparent bg-muted/50'
   469→                      : 'border-muted-foreground/25 hover:border-muted-foreground/50 hover:bg-muted/50'
   470→                }`}
   471→              >
   472→                {form.image ? (
   473→                  <div className="relative w-full">
   474→                    <img
   475→                      src={form.image}
   476→                      alt="Preview"
   477→                      className="mx-auto max-h-48 rounded-lg object-contain"
   478→                    />
   479→                    <button
   480→                      type="button"
   481→                      onClick={(e) => {
   482→                        e.stopPropagation();
   483→                        setForm((f) => ({ ...f, image: null }));
   484→                      }}
   485→                      className="absolute top-2 right-2 flex size-7 items-center justify-center rounded-full bg-black/60 text-white hover:bg-black/80"
   486→                    >
   487→                      <X className="size-3.5" />
   488→                    </button>
   489→                  </div>
   490→                ) : uploading ? (
   491→                  <div className="flex flex-col items-center gap-2">
   492→                    <Loader2 className="size-8 animate-spin text-muted-foreground" />
   493→                    <p className="text-xs text-muted-foreground">Uploading...</p>
   494→                  </div>
   495→                ) : (
   496→                  <div className="flex flex-col items-center gap-2">
   497→                    <div className="flex size-12 items-center justify-center rounded-full bg-muted">
   498→                      <Upload className="size-5 text-muted-foreground" />
   499→                    </div>
   500→                    <div className="text-center">
   501→                      <p className="text-sm font-medium text-foreground">
   502→                        Drop image here or click to browse
   503→                      </p>
   504→                      <p className="mt-0.5 text-xs text-muted-foreground">
   505→                        PNG, JPG, WebP up to 5MB
   506→                      </p>
   507→                    </div>
   508→                  </div>
   509→                )}
   510→                <input
   511→                  ref={fileInputRef}
   512→                  type="file"
   513→                  accept="image/*"
   514→                  className="hidden"
   515→                  onChange={handleFileChange}
   516→                />
   517→              </div>
   518→            </div>
   519→
   520→            {/* Name */}
   521→            <div className="space-y-2">
   522→              <Label htmlFor="item-name">
   523→                Item Name <span className="text-destructive">*</span>
   524→              </Label>
   525→              <Input
   526→                id="item-name"
   527→                placeholder="e.g. Jollof Rice"
   528→                value={form.name}
   529→                onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   530→              />
   531→            </div>
   532→
   533→            {/* Description */}
   534→            <div className="space-y-2">
   535→              <Label htmlFor="item-desc">Description</Label>
   536→              <Textarea
   537→                id="item-desc"
   538→                placeholder="Brief description of the dish..."
   539→                value={form.description}
   540→                onChange={(e) => setForm((f) => ({ ...f, description: e.target.value }))}
   541→                rows={3}
   542→              />
   543→            </div>
   544→
   545→            {/* Price + Category Row */}
   546→            <div className="grid grid-cols-2 gap-4">
   547→              <div className="space-y-2">
   548→                <Label htmlFor="item-price">
   549→                  Price <span className="text-destructive">*</span>
   550→                </Label>
   551→                <div className="relative">
   552→                  <span className="absolute top-1/2 left-3 -translate-y-1/2 text-sm font-medium text-muted-foreground">
   553→                    ₦
   554→                  </span>
   555→                  <Input
   556→                    id="item-price"
   557→                    type="number"
   558→                    min="0"
   559→                    step="1"
   560→                    placeholder="0"
   561→                    value={form.price}
   562→                    onChange={(e) => setForm((f) => ({ ...f, price: e.target.value }))}
   563→                    className="pl-8"
   564→                  />
   565→                </div>
   566→              </div>
   567→              <div className="space-y-2">
   568→                <Label>Category</Label>
   569→                <Select
   570→                  value={form.categoryId}
   571→                  onValueChange={(v) => setForm((f) => ({ ...f, categoryId: v }))}
   572→                >
   573→                  <SelectTrigger className="w-full">
   574→                    <SelectValue placeholder="Select category" />
   575→                  </SelectTrigger>
   576→                  <SelectContent>
   577→                    {categories.map((cat) => (
   578→                      <SelectItem key={cat.id} value={cat.id}>
   579→                        {cat.name}
   580→                      </SelectItem>
   581→                    ))}
   582→                  </SelectContent>
   583→                </Select>
   584→              </div>
   585→            </div>
   586→
   587→            {/* Availability */}
   588→            <div className="flex items-center justify-between rounded-lg border p-3">
   589→              <div className="space-y-0.5">
   590→                <Label className="text-sm font-medium">Available</Label>
   591→                <p className="text-xs text-muted-foreground">
   592→                  Item will be visible on the public menu
   593→                </p>
   594→              </div>
   595→              <Switch
   596→                checked={form.available}
   597→                onCheckedChange={(v) => setForm((f) => ({ ...f, available: v }))}
   598→              />
   599→            </div>
   600→          </div>
   601→        </ScrollArea>
   602→
   603→        <Separator />
   604→
   605→        <DialogFooter className="shrink-0 px-6 py-4">
   606→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   607→            Cancel
   608→          </Button>
   609→          <Button
   610→            onClick={handleSubmit}
   611→            disabled={!form.name.trim() || !form.price || !form.categoryId || loading}
   612→            className="bg-emerald-600 hover:bg-emerald-700"
   613→          >
   614→            {loading && <Loader2 className="size-4 animate-spin" />}
   615→            {isEditing ? 'Save Changes' : 'Add Item'}
   616→          </Button>
   617→        </DialogFooter>
   618→      </DialogContent>
   619→    </Dialog>
   620→  );
   621→}
