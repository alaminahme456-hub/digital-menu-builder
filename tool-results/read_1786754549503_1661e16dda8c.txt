   436→// Main Templates Component
   437→// ──────────────────────────────────────────────
   438→export default function Templates() {
   439→  const { token } = useAuthStore();
   440→  const { currentBusiness, setActiveTemplate, navigate } = useAppStore();
   441→
   442→  // Local UI state
   443→  const [previewTemplate, setPreviewTemplate] = useState<string | null>(null);
   444→  const [applyingTemplate, setApplyingTemplate] = useState<string | null>(null);
   445→  const [failedTemplate, setFailedTemplate] = useState<string | null>(null);
   446→  const [saving, setSaving] = useState(false);
   447→  const [primaryColor, setPrimaryColor] = useState('#10b981');
   448→  const [secondaryColor, setSecondaryColor] = useState('#f1f5f9');
   449→  const [selectedFont, setSelectedFont] = useState('inter');
   450→
   451→  // Track the applied template locally for instant UI updates
   452→  const [localActiveTemplate, setLocalActiveTemplate] = useState<string | null>(null);
   453→  const prevBusinessRef = useRef(currentBusiness);
   454→
   455→  // Sync colors/font from current business
   456→  useEffect(() => {
   457→    if (currentBusiness) {
   458→      setPrimaryColor(currentBusiness.primaryColor || '#10b981');
   459→      setSecondaryColor(currentBusiness.secondaryColor || '#f1f5f9');
   460→      setSelectedFont(currentBusiness.fontFamily || 'inter');
   461→      // Initialize local template from business data
   462→      if (!localActiveTemplate) {
   463→        setLocalActiveTemplate(currentBusiness.templateName || 'modern');
   464→      }
   465→    }
   466→  }, [currentBusiness]);
   467→
   468→  // Reset local template when business changes
   469→  useEffect(() => {
   470→    if (prevBusinessRef.current?.id !== currentBusiness?.id) {
   471→      setLocalActiveTemplate(currentBusiness?.templateName || 'modern');
   472→      prevBusinessRef.current = currentBusiness;
   473→    }
   474→  }, [currentBusiness?.id]);
   475→
   476→  // Current active template: prioritize local state (optimistic), fall back to business
   477→  const currentTemplate = localActiveTemplate || currentBusiness?.templateName || 'modern';
   478→
   479→  // ──────────────────────────────────────────────
   480→  // INSTANT APPLY — Optimistic UI + background save
   481→  // ──────────────────────────────────────────────
   482→  const applyTemplate = useCallback(async (templateName: string) => {
   483→    if (!currentBusiness?.id || !token) return;
   484→    if (currentTemplate === templateName) return;
   485→
   486→    // 1. Instantly update local UI (optimistic)
   487→    const previousTemplate = currentTemplate;
   488→    setLocalActiveTemplate(templateName);
   489→    setFailedTemplate(null);
   490→    setActiveTemplate(templateName);
   491→
   492→    // 2. Start background save
   493→    setApplyingTemplate(templateName);
   494→
   495→    try {
   496→      const res = await fetch(`/api/businesses/${currentBusiness.id}`, {
   497→        method: 'PUT',
   498→        headers: {
   499→          Authorization: `Bearer ${token}`,
   500→          'Content-Type': 'application/json',
   501→        },
   502→        body: JSON.stringify({ templateName }),
   503→      });
   504→
   505→      if (!res.ok) {
   506→        // Revert on failure
   507→        setLocalActiveTemplate(previousTemplate);
   508→        setActiveTemplate(previousTemplate);
   509→        setFailedTemplate(templateName);
   510→        toast.error('Unable to apply template. Please try again.');
   511→      } else {
   512→        const data = await res.json();
   513→        const updatedBusiness = data.business;
   514→
   515→        // Update current business in store with new template info
   516→        if (updatedBusiness) {
   517→          useAppStore.getState().setCurrentBusiness({
   518→            ...currentBusiness,
   519→            templateName: updatedBusiness.templateName || templateName,
   520→            primaryColor: updatedBusiness.primaryColor || currentBusiness.primaryColor,
   521→            secondaryColor: updatedBusiness.secondaryColor || currentBusiness.secondaryColor,
   522→            fontFamily: updatedBusiness.fontFamily || currentBusiness.fontFamily,
   523→          });
   524→        }
   525→
   526→        // Log template application
   527→        try {
   528→          await fetch(`/api/businesses/${currentBusiness.id}/template-log`, {
   529→            method: 'POST',
   530→            headers: {
   531→              Authorization: `Bearer ${token}`,
   532→              'Content-Type': 'application/json',
   533→            },
   534→            body: JSON.stringify({
   535→              templateId: templateName,
   536→              businessId: currentBusiness.id,
   537→            }),
   538→          });
   539→        } catch {
   540→          // Logging failure is non-critical, silently ignore
   541→        }
   542→
   543→        toast.success(`"${TEMPLATES.find(t => t.name === templateName)?.label}" template applied!`, {
   544→          description: 'Your menu preview has been updated instantly.',
   545→          duration: 3000,
   546→        });
   547→      }
   548→    } catch {
   549→      // Network error — revert
   550→      setLocalActiveTemplate(previousTemplate);
   551→      setActiveTemplate(previousTemplate);
   552→      setFailedTemplate(templateName);
   553→      toast.error('Unable to apply template. Please try again.');
   554→    } finally {
   555→      setApplyingTemplate(null);
   556→    }
   557→  }, [currentBusiness, token, currentTemplate, setActiveTemplate]);
   558→
   559→  // ──────────────────────────────────────────────
   560→  // Customization save
   561→  // ──────────────────────────────────────────────
   562→  const applyCustomization = async () => {
   563→    if (!currentBusiness?.id || !token) return;
   564→    setSaving(true);

   567→        method: 'PUT',
   568→        headers: {
   569→          Authorization: `Bearer ${token}`,
   570→          'Content-Type': 'application/json',
   571→        },
   572→        body: JSON.stringify({ primaryColor, secondaryColor, fontFamily: selectedFont }),
   573→      });
   574→
   575→      if (res.ok) {
   576→        const data = await res.json();
   577→        const updatedBusiness = data.business;
   578→        if (updatedBusiness) {
   579→          useAppStore.getState().setCurrentBusiness({
   580→            ...currentBusiness,
   581→            primaryColor: updatedBusiness.primaryColor,
   582→            secondaryColor: updatedBusiness.secondaryColor,
   583→            fontFamily: updatedBusiness.fontFamily,
   584→          });
   585→        }
   586→        toast.success('Customization applied successfully!');
   587→      } else {
   588→        toast.error('Failed to apply changes');
   589→      }
   590→    } catch {
   591→      toast.error('Failed to apply changes');
   592→    } finally {
   593→      setSaving(false);
   594→    }
   595→  };

   622→            <Sparkles className="h-5 w-5 text-gold" />
   623→            Design Templates

   626→            Choose a template and apply it instantly. Your menu preview updates in real-time.

   642→        {TEMPLATES.map((template) => (

   655→      <Card className="mt-6">
   656→        <CardHeader>
   657→          <CardTitle className="flex items-center gap-2 text-lg">
   658→            <Paintbrush className="h-5 w-5 text-gold" />
   659→            Customize Design
   660→          </CardTitle>
   661→          <CardDescription>
   662→            Fine-tune your menu&apos;s colors and typography. Changes apply to the current template.
   663→          </CardDescription>
   664→        </CardHeader>
   665→        <CardContent>
   666→          <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
