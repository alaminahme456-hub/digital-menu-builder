     1→'use client';
     2→
     3→import { useState, useEffect, useRef } from 'react';
     4→import {
     5→  Save,
     6→  Upload,
     7→  Trash2,
     8→  Loader2,
     9→  ImageIcon,
    10→  AlertTriangle,
    11→} from 'lucide-react';
    12→import {
    13→  Card,
    14→  CardContent,
    15→  CardHeader,
    16→  CardTitle,
    17→  CardDescription,
    18→} from '@/components/ui/card';
    19→import { Button } from '@/components/ui/button';
    20→import { Input } from '@/components/ui/input';
    21→import { Textarea } from '@/components/ui/textarea';
    22→import { Label } from '@/components/ui/label';
    23→import { Switch } from '@/components/ui/switch';
    24→import { Separator } from '@/components/ui/separator';
    25→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    26→import {
    27→  Select,
    28→  SelectContent,
    29→  SelectItem,
    30→  SelectTrigger,
    31→  SelectValue,
    32→} from '@/components/ui/select';
    33→import {
    34→  AlertDialog,
    35→  AlertDialogAction,
    36→  AlertDialogCancel,
    37→  AlertDialogContent,
    38→  AlertDialogDescription,
    39→  AlertDialogFooter,
    40→  AlertDialogHeader,
    41→  AlertDialogTitle,
    42→  AlertDialogTrigger,
    43→} from '@/components/ui/alert-dialog';
    44→import { useAuthStore, useAppStore } from '@/lib/store';
    45→import type { Business } from '@/lib/types';
    46→import { BUSINESS_CATEGORIES } from '@/lib/types';
    47→import { getAppUrl } from '@/lib/auth';
    48→import { toast } from 'sonner';
    49→
    50→interface SettingsPanelProps {
    51→  initialTab?: string;
    52→}
    53→
    54→export default function SettingsPanel({ initialTab = 'business' }: SettingsPanelProps) {
    55→  const { token, user } = useAuthStore();
    56→  const { currentBusiness, setCurrentBusiness } = useAppStore();
    57→  const fileInputRef = useRef<HTMLInputElement>(null);
    58→
    59→  // Business form state
    60→  const [bizForm, setBizForm] = useState({
    61→    name: '',
    62→    category: '',
    63→    phone: '',
    64→    whatsapp: '',
    65→    address: '',
    66→    openingHours: '',
    67→    description: '',
    68→    status: 'draft' as string,
    69→    whatsappOrder: false,
    70→    seoEnabled: false,
    71→  });
    72→  const [bizLoading, setBizLoading] = useState(false);
    73→  const [bizSaving, setBizSaving] = useState(false);
    74→  const [bizInitial, setBizInitial] = useState<Business | null>(null);
    75→
    76→  // Account form state
    77→  const [accForm, setAccForm] = useState({
    78→    name: '',
    79→    phone: '',
    80→    currentPassword: '',
    81→    newPassword: '',
    82→    confirmPassword: '',
    83→  });
    84→  const [accSaving, setAccSaving] = useState(false);
    85→  const [accLoading, setAccLoading] = useState(true);
    86→
    87→  // Upload state
    88→  const [uploading, setUploading] = useState(false);
    89→
    90→  // Flipbook settings state
    91→  const [flipbookForm, setFlipbookForm] = useState({
    92→    flipbookEnabled: true,
    93→    flipbookAnimEnabled: true,
    94→    flipbookAnimSpeed: 'medium' as string,
    95→    flipbookPageNumbers: true,
    96→    flipbookSwipeNav: true,
    97→    flipbookSoundEffects: false,
    98→    flipbookFullscreen: true,
    99→    flipbookInteractions: true,
   100→  });
   101→  const [flipbookSaving, setFlipbookSaving] = useState(false);
   102→
   103→  // Ordering settings state
   104→  const [orderingForm, setOrderingForm] = useState({
   105→    whatsappOrder: true,
   106→    basketEnabled: true,
   107→    showQuantitySelector: true,
   108→    showOrderButton: true,
   109→    whatsappGreeting: 'Hello, I would like to place an order:',
   110→  });
   111→  const [orderingSaving, setOrderingSaving] = useState(false);
   112→
   113→  // Fetch business data
   114→  useEffect(() => {
   115→    if (!currentBusiness?.id || !token) return;
   116→    setBizLoading(true);
   117→    async function fetchBusiness() {
   118→      try {
   119→        const res = await fetch(`/api/businesses/${currentBusiness.id}`, {
   120→          headers: { Authorization: `Bearer ${token}` },
   121→        });
   122→        if (res.ok) {
   123→          const data = await res.json();
   124→          const b: Business = data.business;
   125→          setBizInitial(b);
   126→          setBizForm({
   127→            name: b.name || '',
   128→            category: b.category || '',
   129→            phone: b.phone || '',
   130→            whatsapp: b.whatsapp || '',
   131→            address: b.address || '',
   132→            openingHours: b.openingHours || '',
   133→            description: b.description || '',
   134→            status: b.status || 'draft',
   135→            whatsappOrder: b.whatsappOrder ?? false,
   136→            seoEnabled: b.seoEnabled ?? false,
   137→            welcomeMessage: (b as Record<string, unknown>).welcomeMessage as string || '',
   138→          });
   139→          // Load flipbook settings
   140→          setFlipbookForm({
   141→            flipbookEnabled: b.flipbookEnabled ?? true,
   142→            flipbookAnimEnabled: b.flipbookAnimEnabled ?? true,
   143→            flipbookAnimSpeed: b.flipbookAnimSpeed || 'medium',
   144→            flipbookPageNumbers: b.flipbookPageNumbers ?? true,
   145→            flipbookSwipeNav: b.flipbookSwipeNav ?? true,
   146→            flipbookSoundEffects: b.flipbookSoundEffects ?? false,
   147→            flipbookFullscreen: b.flipbookFullscreen ?? true,
   148→            flipbookInteractions: b.flipbookInteractions ?? true,
   149→          });
   150→          // Load ordering settings
   151→          setOrderingForm({
   152→            whatsappOrder: b.whatsappOrder ?? true,
   153→            basketEnabled: b.basketEnabled ?? true,
   154→            showQuantitySelector: b.showQuantitySelector ?? true,
   155→            showOrderButton: b.showOrderButton ?? true,
   156→            whatsappGreeting: b.whatsappGreeting || 'Hello, I would like to place an order:',
   157→          });
   158→        }
   159→      } catch {
   160→        toast.error('Failed to load business settings');
   161→      } finally {
   162→        setBizLoading(false);
   163→      }
   164→    }
   165→    fetchBusiness();
   166→  }, [currentBusiness?.id, token]);
   167→
   168→  // Fetch user data
   169→  useEffect(() => {
   170→    if (!user || !token) return;
   171→    setAccForm({
   172→      name: user.name || '',
   173→      phone: '',
   174→      currentPassword: '',
   175→      newPassword: '',
   176→      confirmPassword: '',
   177→    });
   178→    setAccLoading(false);
   179→  }, [user, token]);
   180→
   181→  // Save business settings
   182→  const handleSaveBusiness = async () => {
   183→    if (!currentBusiness?.id || !token) return;
   184→    setBizSaving(true);
   185→    try {
   186→      const res = await fetch(`/api/businesses/${currentBusiness.id}`, {
   187→        method: 'PUT',
   188→        headers: {
   189→          Authorization: `Bearer ${token}`,
   190→          'Content-Type': 'application/json',
   191→        },
   192→        body: JSON.stringify(bizForm),
   193→      });
   194→      if (res.ok) {
   195→        const data = await res.json();
   196→        const updated: Business = data.business;
   197→        setBizInitial(updated);
   198→        setCurrentBusiness({
   199→          id: updated.id,
   200→          slug: updated.slug,
   201→          name: updated.name,
   202→          logo: updated.logo,
   203→          status: updated.status,
   204→        });
   205→        toast.success('Business settings saved');
   206→      } else {
   207→        const err = await res.json().catch(() => ({}));
   208→        toast.error(err.error || 'Failed to save settings');
   209→      }
   210→    } catch {
   211→      toast.error('Failed to save settings');
   212→    } finally {
   213→      setBizSaving(false);
   214→    }
   215→  };
   216→
   217→  // Save account settings
   218→  const handleSaveAccount = async () => {
   219→    if (!token) return;
   220→    const updateData: Record<string, string> = {};
   221→    if (accForm.name) updateData.name = accForm.name;
   222→    if (accForm.phone) updateData.phone = accForm.phone;
   223→
   224→    // Password change validation
   225→    if (accForm.newPassword) {
   226→      if (!accForm.currentPassword) {
   227→        toast.error('Current password is required to change password');
   228→        return;
   229→      }
   230→      if (accForm.newPassword !== accForm.confirmPassword) {
   231→        toast.error('New passwords do not match');
   232→        return;
   233→      }
   234→      if (accForm.newPassword.length < 8) {
   235→        toast.error('Password must be at least 8 characters');
   236→        return;
   237→      }
   238→      updateData.currentPassword = accForm.currentPassword;
   239→      updateData.newPassword = accForm.newPassword;
   240→    }
   241→
   242→    if (Object.keys(updateData).length === 0) {
   243→      toast.error('No changes to save');
   244→      return;
   245→    }
   246→
   247→    setAccSaving(true);
   248→    try {
   249→      const res = await fetch('/api/auth/me', {
   250→        method: 'PUT',
   251→        headers: {
   252→          Authorization: `Bearer ${token}`,
   253→          'Content-Type': 'application/json',
   254→        },
   255→        body: JSON.stringify(updateData),
   256→      });
   257→      if (res.ok) {
   258→        toast.success('Account settings saved');
   259→        setAccForm((prev) => ({
   260→          ...prev,
   261→          currentPassword: '',
   262→          newPassword: '',
   263→          confirmPassword: '',
   264→        }));
   265→      } else {
   266→        const err = await res.json().catch(() => ({}));
   267→        toast.error(err.error || 'Failed to save account');
   268→      }
   269→    } catch {
   270→      toast.error('Failed to save account');
   271→    } finally {
   272→      setAccSaving(false);
   273→    }
   274→  };
   275→
   276→  // Logo upload
   277→  const handleLogoUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   278→    const file = e.target.files?.[0];
   279→    if (!file || !currentBusiness?.id || !token) return;
   280→    setUploading(true);
   281→    try {
   282→      const formData = new FormData();
   283→      formData.append('file', file);
   284→      formData.append('businessId', currentBusiness.id);
   285→      formData.append('type', 'logo');
   286→      const res = await fetch('/api/upload', {
   287→        method: 'POST',
   288→        headers: { Authorization: `Bearer ${token}` },
   289→        body: formData,
   290→      });
   291→      if (res.ok) {
   292→        const data = await res.json();
   293→        if (data.url) {
   294→          setCurrentBusiness({
   295→            ...currentBusiness,
   296→            logo: data.url,
   297→          });
   298→          toast.success('Logo uploaded');
   299→        }
   300→      } else {
   301→        toast.error('Failed to upload logo');
   302→      }
   303→    } catch {
   304→      toast.error('Failed to upload logo');
   305→    } finally {
   306→      setUploading(false);
   307→      if (fileInputRef.current) fileInputRef.current.value = '';
   308→    }
   309→  };
   310→
   311→  // Delete account
   312→  const handleDeleteAccount = async () => {
   313→    if (!token) return;
   314→    try {
   315→      const res = await fetch('/api/auth/me', {
   316→        method: 'DELETE',
   317→        headers: { Authorization: `Bearer ${token}` },
   318→      });
   319→      if (res.ok) {
   320→        toast.success('Account deleted');
   321→        useAuthStore.getState().logout();
   322→      } else {
   323→        toast.error('Failed to delete account');
   324→      }
   325→    } catch {
   326→      toast.error('Failed to delete account');
   327→    }
   328→  };
   329→
   330→  if (!currentBusiness) {
   331→    return (
   332→      <div className="flex flex-col items-center justify-center py-20 gap-3 text-muted-foreground">
   333→        <p>Select a business to manage settings</p>
   334→      </div>
   335→    );
   336→  }
   337→
   338→  const hasBizChanges =
   339→    bizInitial &&
   340→    (bizForm.name !== bizInitial.name ||
   341→      bizForm.category !== (bizInitial.category || '') ||
   342→      bizForm.phone !== (bizInitial.phone || '') ||
   343→      bizForm.whatsapp !== (bizInitial.whatsapp || '') ||
   344→      bizForm.address !== (bizInitial.address || '') ||
   345→      bizForm.openingHours !== (bizInitial.openingHours || '') ||
   346→      bizForm.description !== (bizInitial.description || '') ||
   347→      bizForm.status !== bizInitial.status ||
   348→      bizForm.whatsappOrder !== (bizInitial.whatsappOrder ?? false) ||
   349→      bizForm.seoEnabled !== (bizInitial.seoEnabled ?? false));
   350→
   351→return (
   352→    <>
   353→    <Tabs defaultValue={initialTab} className="space-y-6">
   354→      <TabsList>
   355→        <TabsTrigger value="business">Business Settings</TabsTrigger>
   356→        <TabsTrigger value="flipbook">Flipbook</TabsTrigger>
   357→        <TabsTrigger value="ordering">Ordering</TabsTrigger>
   358→        <TabsTrigger value="account">Account Settings</TabsTrigger>
   359→      </TabsList>
   360→
   361→      {/* Business Settings Tab */}
   362→      <TabsContent value="business" className="space-y-6">
   363→        {/* Basic Info */}
   364→        <Card>
   365→          <CardHeader>
   366→            <CardTitle>Basic Information</CardTitle>
   367→            <CardDescription>
   368→              Update your business details and branding
   369→            </CardDescription>
   370→          </CardHeader>
   371→          <CardContent className="space-y-4">
   372→            {bizLoading ? (
   373→              <div className="space-y-4">
   374→                {[1, 2, 3, 4].map((i) => (
   375→                  <div key={i} className="space-y-2">
   376→                    <div className="h-4 w-24 rounded bg-muted" />
   377→                    <div className="h-9 rounded-md bg-muted" />
   378→                  </div>
   379→                ))}
   380→              </div>
   381→            ) : (
   382→              <>
   383→                {/* Logo Upload */}
   384→                <div className="space-y-2">
   385→                  <Label>Logo</Label>
   386→                  <div className="flex items-center gap-4">
   387→                    {currentBusiness.logo ? (
   388→                      <img
   389→                        src={currentBusiness.logo}
   390→                        alt="Logo"
   391→                        className="h-16 w-16 rounded-xl object-cover ring-1 ring-border"
   392→                      />
   393→                    ) : (
   394→                      <div className="flex h-16 w-16 items-center justify-center rounded-xl bg-muted">
   395→                        <ImageIcon className="h-6 w-6 text-muted-foreground" />
   396→                      </div>
   397→                    )}
   398→                    <div className="flex flex-col gap-2">
   399→                      <Button
   400→                        type="button"
   401→                        variant="outline"
   402→                        size="sm"
   403→                        onClick={() => fileInputRef.current?.click()}
   404→                        disabled={uploading}
   405→                      >
   406→                        {uploading ? (
   407→                          <Loader2 className="mr-1.5 h-4 w-4 animate-spin" />
   408→                        ) : (
   409→                          <Upload className="mr-1.5 h-4 w-4" />
   410→                        )}
   411→                        {uploading ? 'Uploading...' : 'Upload Logo'}
   412→                      </Button>
   413→                      <input
   414→                        ref={fileInputRef}
   415→                        type="file"
   416→                        accept="image/*"
   417→                        onChange={handleLogoUpload}
   418→                        className="hidden"
   419→                      />
   420→                      <p className="text-xs text-muted-foreground">
   421→                        PNG, JPG up to 2MB
   422→                      </p>
   423→                    </div>
   424→                  </div>
   425→                </div>
   426→
   427→                <Separator />
   428→
   429→                {/* Business Name */}
   430→                <div className="space-y-2">
   431→                  <Label htmlFor="biz-name">Business Name</Label>
   432→                  <Input
   433→                    id="biz-name"
   434→                    value={bizForm.name}
   435→                    onChange={(e) =>
   436→                      setBizForm((p) => ({ ...p, name: e.target.value }))
   437→                    }
   438→                    placeholder="Enter business name"
   439→                  />
   440→                </div>
   441→
   442→                {/* Public URL / Slug */}
   443→                <div className="space-y-2">
   444→                  <Label htmlFor="biz-slug">Public URL</Label>
   445→                  <div className="flex items-center gap-0">
   446→                    <div className="flex items-center bg-muted text-muted-foreground text-sm px-3 py-2 rounded-l-md border border-r-0 border-input">
   447→                      {getAppUrl()}/p/
   448→                    </div>
   449→                    <Input
   450→                      id="biz-slug"
   451→                      value={currentBusiness?.slug || ''}
   452→                      disabled
   453→                      className="rounded-l-none"
   454→                      placeholder="auto-generated-slug"
   455→                    />
   456→                  </div>
   457→                  <p className="text-xs text-muted-foreground">
   458→                    Slug is auto-generated from your business name. Contact support to change it.
   459→                  </p>
   460→                </div>
   461→
   462→                {/* Category */}
   463→                <div className="space-y-2">
   464→                  <Label>Category</Label>
   465→                  <Select
   466→                    value={bizForm.category}
   467→                    onValueChange={(v) =>
   468→                      setBizForm((p) => ({ ...p, category: v }))
   469→                    }
   470→                  >
   471→                    <SelectTrigger className="w-full">
   472→                      <SelectValue placeholder="Select a category" />
   473→                    </SelectTrigger>
   474→                    <SelectContent>
   475→                      {BUSINESS_CATEGORIES.map((cat) => (
   476→                        <SelectItem key={cat} value={cat}>
   477→                          {cat}
   478→                        </SelectItem>
   479→                      ))}
   480→                    </SelectContent>
   481→                  </Select>
   482→                </div>
   483→
   484→                {/* Phone & WhatsApp */}
   485→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   486→                  <div className="space-y-2">
   487→                    <Label htmlFor="biz-phone">Phone</Label>
   488→                    <Input
   489→                      id="biz-phone"
   490→                      value={bizForm.phone}
   491→                      onChange={(e) =>
   492→                        setBizForm((p) => ({ ...p, phone: e.target.value }))
   493→                      }
   494→                      placeholder="Phone number"
   495→                    />
   496→                  </div>
   497→                  <div className="space-y-2">
   498→                    <Label htmlFor="biz-whatsapp">WhatsApp</Label>
   499→                    <Input
   500→                      id="biz-whatsapp"
   501→                      value={bizForm.whatsapp}
   502→                      onChange={(e) =>
   503→                        setBizForm((p) => ({ ...p, whatsapp: e.target.value }))
   504→                      }
   505→                      placeholder="WhatsApp number"
   506→                    />
   507→                  </div>
   508→                </div>
   509→
   510→                {/* Address */}
   511→                <div className="space-y-2">
   512→                  <Label htmlFor="biz-address">Address</Label>
   513→                  <Input
   514→                    id="biz-address"
   515→                    value={bizForm.address}
   516→                    onChange={(e) =>
   517→                      setBizForm((p) => ({ ...p, address: e.target.value }))
   518→                    }
   519→                    placeholder="Business address"
   520→                  />
   521→                </div>
   522→
   523→                {/* Opening Hours */}
   524→                <div className="space-y-2">
   525→                  <Label htmlFor="biz-hours">Opening Hours</Label>
   526→                  <Input
   527→                    id="biz-hours"
   528→                    value={bizForm.openingHours}
   529→                    onChange={(e) =>
   530→                      setBizForm((p) => ({ ...p, openingHours: e.target.value }))
   531→                    }
   532→                    placeholder="e.g. Mon-Fri 9AM-10PM, Sat-Sun 10AM-11PM"
   533→                  />
   534→                </div>
   535→
   536→                {/* Description */}
   537→                <div className="space-y-2">
   538→                  <Label htmlFor="biz-desc">Description / Tagline</Label>
   539→                  <Textarea
   540→                    id="biz-desc"
   541→                    value={bizForm.description}
   542→                    onChange={(e) =>
   543→                      setBizForm((p) => ({ ...p, description: e.target.value }))
   544→                    }
   545→                    placeholder="Brief description of your business"
   546→                    rows={3}
   547→                  />
   548→                </div>
   549→
   550→                {/* Welcome Message */}
   551→                <div className="space-y-2">
   552→                  <Label htmlFor="biz-welcome">Welcome Message</Label>
   553→                  <Textarea
   554→                    id="biz-welcome"
   555→                    value={(bizForm as Record<string, unknown>).welcomeMessage as string || ''}
   556→                    onChange={(e) =>
   557→                      setBizForm((p) => ({ ...p, welcomeMessage: e.target.value }))
   558→                    }
   559→                    placeholder="Welcome message shown to customers when they open your menu. e.g. Thank you for visiting us. Explore our menu and discover our products."
   560→                    rows={3}
   561→                  />
   562→                  <p className="text-xs text-muted-foreground">
   563→                    This message appears on the Welcome page of your digital book experience.
   564→                  </p>
   565→                </div>
   566→              </>
   567→            )}
   568→          </CardContent>
   569→        </Card>
   570→
   571→        {/* Menu Settings */}
   572→        <Card>
   573→          <CardHeader>
   574→            <CardTitle>Menu Settings</CardTitle>
   575→            <CardDescription>
   576→              Control how your menu is published and accessed
   577→            </CardDescription>
   578→          </CardHeader>
   579→          <CardContent className="space-y-6">
   580→            {bizLoading ? (
   581→              <div className="space-y-4">
   582→                {[1, 2, 3].map((i) => (
   583→                  <div key={i} className="flex items-center justify-between">
   584→                    <div className="space-y-1">
   585→                      <div className="h-4 w-32 rounded bg-muted" />
   586→                      <div className="h-3 w-48 rounded bg-muted" />
   587→                    </div>
   588→                    <div className="h-5 w-10 rounded-full bg-muted" />
   589→                  </div>
   590→                ))}
   591→              </div>
   592→            ) : (
   593→              <>
   594→                {/* Menu Status */}
   595→                <div className="space-y-2">
   596→                  <Label>Menu Status</Label>
   597→                  <Select
   598→                    value={bizForm.status}
   599→                    onValueChange={(v) =>
   600→                      setBizForm((p) => ({ ...p, status: v }))
   601→                    }
   602→                  >
   603→                    <SelectTrigger className="w-full sm:w-64">
   604→                      <SelectValue />
   605→                    </SelectTrigger>
   606→                    <SelectContent>
   607→                      <SelectItem value="draft">Draft</SelectItem>
   608→                      <SelectItem value="published">Published</SelectItem>
   609→                      <SelectItem value="unpublished">Unpublished</SelectItem>
   610→                    </SelectContent>
   611→                  </Select>
   612→                  <p className="text-xs text-muted-foreground">
   613→                    Only published menus are accessible via QR code
   614→                  </p>
   615→                </div>
   616→
   617→                <Separator />
   618→
   619→                {/* WhatsApp Ordering */}
   620→                <div className="flex items-center justify-between gap-4">
   621→                  <div className="space-y-0.5">
